package EgasArch_Needs_Concern {
    private import EgasArch_Common_Metadata::*;
    private import EgasArch_Obj_Objective::*;
    private import EgasArch_Needs_Stakeholder::*;

    // hard concerns refine SafeTorqueObjective, soft concerns refine ResponsiveTorqueObjective
    concern def UnintendedAccelerationConcern :> SafeTorqueObjective {
        doc
        /* Occupants and the safety regulator are concerned that the engine accelerates without a driver request. */
        subject egasSystem;
        stakeholder driver : Driver;
        stakeholder safetyRegulator : SafetyRegulator;
    }

    concern def HardwareFailureConcern :> SafeTorqueObjective {
        doc
        /* A sensor or actuator fault shall not cause unintended acceleration; a safe state shall be guaranteed. */
        subject egasSystem;
        stakeholder safetyRegulator : SafetyRegulator;
    }

    concern def FunctionalSafetyConcern :> SafeTorqueObjective {
        doc
        /* Compliance with the functional safety standard shall be demonstrated so that faults do not injure people. */
        subject egasSystem;
        stakeholder safetyRegulator : SafetyRegulator;
    }

    concern def TorqueArbitrationConcern :> SafeTorqueObjective {
        doc
        /* Conflicting torque requests and simultaneous brake and accelerator operation shall not cost the driver control of the vehicle. */
        subject egasSystem;
        stakeholder driver : Driver;
        stakeholder platformEngineer : PlatformEngineer;
    }

    concern def PreciseControlConcern :> ResponsiveTorqueObjective {
        doc
        /* The error between the requested torque and the engine torque shall be small and without delay. */
        subject egasSystem;
        stakeholder driver : Driver;
    }

    concern def DrivabilityAndResponseConcern :> ResponsiveTorqueObjective {
        doc
        /* The vehicle shall respond to pedal input immediately and predictably, without jerk. */
        subject egasSystem;
        stakeholder driver : Driver;
    }

    #GoalConflict dependency arbitrationConflictsDrivability
        from TorqueArbitrationConcern
        to DrivabilityAndResponseConcern {
        doc
        /* Brake override limits output at once and causes jerk. Safety wins; the jerk criterion is waived during the intervention. */
    }
    #GoalConflict dependency hardwareFailureConflictsDrivability
        from HardwareFailureConcern
        to DrivabilityAndResponseConcern {
        doc
        /* A fault reaction cuts air and fuel at once and causes jerk. Safety wins once a critical fault is confirmed. */
    }
    #GoalContribution dependency preciseControlSupportsDrivability
        from PreciseControlConcern
        to DrivabilityAndResponseConcern {
        doc
        /* Less steady-state error and overshoot also reduce jerk during acceleration and deceleration. */
    }
} // package EgasArch_Needs_Concern

package EgasArch_Beh_Behavior {
    private import ModelingMetadata::*;
    private import EgasArch_Common_Enums::*;
    private import EgasArch_Common_Units::TorqueQuantity;
    private import EgasArch_Common_Signals::*;
    private import EgasArch_Shr_Requirement::*;
    private import EgasArch_Scn_Scenario::*;

    item def TorqueDemand {
        doc
        /* Internal absolute torque demand. It differs from the relative TorqueInterventionRequest at the system boundary. */
        attribute requestedTorque : TorqueQuantity;
    }

    action def ArbitrateTorque {
        doc
        /* Turns the driver request into the torque demand. Cruise control and other demand sources are out of the pilot scope. */
        in pedal : AccelerationPedalPosition;
        out demand : TorqueDemand;
    }

    action def ReduceTorque {
        doc
        /* Reduces the demand while an external intervention request or the brake is active. The reduction amount is tbd. */
        in request : TorqueInterventionRequest;
        in brake : BrakePedalPosition;
        in demand : TorqueDemand;
        out reducedDemand : TorqueDemand;
    }

    action def LimitTorque {
        doc
        /* Limits the demand to a reduced torque in limp home (EGAS 11.1.1). The limit value is the pilotAssumed catalog entry limpHomeTorqueLimit (OI-10). */
        in demand : TorqueDemand;
        out limitedDemand : TorqueDemand;
    }

    action def OutputTorqueCommand {
        doc
        /* Outputs the air and the fuel commands for the torque demand. */
        in demand : TorqueDemand;
        in engineSensor : EngineSensorData;
        out airFlow : IntakeAirFlow;
        out fuelCommand : FuelIgnitionCommand;
    }

    state def SafeState;
    state def TorqueLimited :> SafeState; // 1단계: 제어 가능한 출력/토크 제한 주행
    state def FuelCutOff :> SafeState; // 2단계: 추진력 차단 (연료 분사 차단 등)
    state def FullCutOff :> SafeState; // 3단계: 출력단 완전 차단 / 엔진 정지 / 시동 차단

    // FSR/TSR Level Fault Reaction

    // 1단계 매핑
    state def SensorLimpHome :> TorqueLimited;
    state def FunctionInhibition :> TorqueLimited;
    state def IdleSpeedDemand :> TorqueLimited;

    // 2단계 매핑
    state def IrreversibleIco :> FuelCutOff;

    // 3단계 매핑
    state def EcuReset :> FullCutOff;
    state def PowerStageShutoff :> FullCutOff;
    state def RemainInBootBlock :> FullCutOff;
    state def SafeEngineStopT15 :> FullCutOff;

    state def EgasBehavior {
        doc
        /* One state machine for the pilot scope. The top-level states are the operating modes; calibration, diagnostics, cruise and latch regions are out of scope. */
        attribute mode : OperatingMode;

        first start then normal;

        state normal {
            doc
            /* Engine runs and the driver request is served; the L2 monitor compares actual and permissible torque. */
            do action arbitrating : ArbitrateTorque;
        }
        state excessSuspected {
            doc
            /* A torque excess is suspected and not yet confirmed. The confirmation time is tbd. */
        }
        state intervention {
            doc
            /* Torque is reduced by an external request or by the brake. */
            do action reducing : ReduceTorque;
        }
        state limpHome {
            doc
            /* Both pedal value sensor channels have failed; torque stays limited. */
            do action limiting : LimitTorque;
        }
        state safeState {
            doc
            /* A torque excess is confirmed and the safe state is reached. The stage value is not fixed here. */
            // attribute stage : SafeStateStage;
            entry action warnDriver { out warning : DashboardWarning; }
        }

        transition toExcessSuspected first normal then excessSuspected;
        transition toSafeState first excessSuspected then safeState;
        transition toInterventionByRequest
            first normal accept request : TorqueInterventionRequest then intervention;
        transition toInterventionByBrake
            first normal accept brake : BrakePedalPosition then intervention;
        transition backToNormal first intervention then normal;
        // The fault condition of the pedal value sensor channels is tbd (OI-12).
        transition toLimpHome first normal accept pedal : AccelerationPedalPosition then limpHome;
    }

    state l2Behavior : EgasBehavior;

    #refinement dependency behaviorRefinesShr from EgasBehavior to MonitorTorqueRequirement;
    #refinement dependency behaviorRefinesMonitorScenario
        from EgasBehavior
        to L2TorqueMonitorScenario;
    #refinement dependency behaviorRefinesLimpHomeScenario
        from EgasBehavior
        to PvsFaultLimpHomeScenario;
    #refinement dependency arbitrateRefinesResponseShr
        from ArbitrateTorque
        to PedalResponseRequirement;
    #refinement dependency arbitrateRefinesDemandScenario
        from ArbitrateTorque
        to DriverTorqueDemandScenario;
    #refinement dependency demandRefinesPrecisionShr
        from TorqueDemand
        to SteadyStatePrecisionRequirement;
    #refinement dependency reduceRefinesInterventionShr
        from ReduceTorque
        to TorqueInterventionRequirement;
    #refinement dependency reduceRefinesBrakeShr from ReduceTorque to BrakeOverrideRequirement;
    #refinement dependency reduceRefinesInterventionScenario
        from ReduceTorque
        to TorqueInterventionScenario;
    #refinement dependency reduceRefinesBrakeScenario from ReduceTorque to BrakeOverrideScenario;
    #refinement dependency limitRefinesLimpHomeScenario
        from LimitTorque
        to PvsFaultLimpHomeScenario;
    #refinement dependency outputRefinesResponseShr
        from OutputTorqueCommand
        to PedalResponseRequirement;
} // package EgasArch_Beh_Behavior

    package AbstractNavigationSteps {
        abstract calc def <Step> NavigationStep {
            doc
            /* The most generic navigation step: element in, collection out. */
            in elements [0..1];
            return targets [0..*]; // empty = navigation had no result
        }
        abstract calc def <Single> SinglevaluedStep :> Step {
            doc
            /* At most one target per input element, which may feed a further
             * step.
             */
        }
        abstract calc def <FanOut> FanOutStep :> Step {
            doc
            /* A step whose targets split the parent row into nested rows — the
             * 1:N join of the table model.
             *
             * A `*Fan` call must be the outermost call of a named view-level ref
             * (an *anchor*), never inline in a column or nested in another call:
             *
             *     ports = UsageFan(baseViewElement, UsageKind::portUsage);
             *
             * An anchor starts from `baseViewElement` or one other anchor, and
             * feeds at most one further anchor — anchors form a chain, not a
             * tree. A column then references one anchor and applies only non-Fan
             * steps to it. Declaration order does not matter, and columns sharing
             * an anchor are row-aligned.
             */
        }
        abstract calc def <List> ListElements :> Step {
            doc
            /* A terminal step: every target rendered together in one cell. Its
             * elements cannot feed another step, so a `*List` call may only be a
             * chain's LAST step. Such a column is read-only for every row.
             */
        }
        abstract calc def <First> FirstStep :> Single {
            doc
            /* For navigations that yield one target under the supported
             * modelling pattern, even though SysML permits more —
             * `DefinitionsFirst` for a usage typed by exactly one definition.
             * Not a way to take the head of a list: if more targets exist the
             * model deviates from the pattern, and the cell renders the first one
             * read-only, with a note.
             */
        }
    } // package AbstractNavigationSteps

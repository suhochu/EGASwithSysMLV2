##egas2/model/01_Requirements/02_1_EgasArch_Needs_Concern.sysml##
package EgasArch_Needs_Concern {
    private import EgasArch_Common_Metadata::*;
    private import EgasArch_Obj_Objective::*;
    private import EgasArch_Needs_Stakeholder::*;

    // hard concerns refine SafeTorqueObjective, soft concerns refine ResponsiveTorqueObjective
    concern def UnintendedAccelerationConcern :> SafeTorqueObjective {
        doc
        /* Occupants and the safety regulator are concerned that the engine accelerates without a driver request. */
        subject egasSystem;
        stakeholder driver : Driver;
        stakeholder safetyRegulator : SafetyRegulator;
    }

    concern def HardwareFailureConcern :> SafeTorqueObjective {
        doc
        /* A sensor or actuator fault shall not cause unintended acceleration; a safe state shall be guaranteed. */
        subject egasSystem;
        stakeholder safetyRegulator : SafetyRegulator;
    }

    concern def FunctionalSafetyConcern :> SafeTorqueObjective {
        doc
        /* Compliance with the functional safety standard shall be demonstrated so that faults do not injure people. */
        subject egasSystem;
        stakeholder safetyRegulator : SafetyRegulator;
    }

    concern def TorqueArbitrationConcern :> SafeTorqueObjective {
        doc
        /* Conflicting torque requests and simultaneous brake and accelerator operation shall not cost the driver control of the vehicle. */
        subject egasSystem;
        stakeholder driver : Driver;
        stakeholder platformEngineer : PlatformEngineer;
    }

    concern def PreciseControlConcern :> ResponsiveTorqueObjective {
        doc
        /* The error between the requested torque and the engine torque shall be small and without delay. */
        subject egasSystem;
        stakeholder driver : Driver;
    }

    concern def DrivabilityAndResponseConcern :> ResponsiveTorqueObjective {
        doc
        /* The vehicle shall respond to pedal input immediately and predictably, without jerk. */
        subject egasSystem;
        stakeholder driver : Driver;
    }

    // Concern usages referenced as #original endpoints in #derivation connections (SG / SHR)
    concern unintendedAccelerationConcern : UnintendedAccelerationConcern;
    concern hardwareFailureConcern : HardwareFailureConcern;
    concern functionalSafetyConcern : FunctionalSafetyConcern;
    concern torqueArbitrationConcern : TorqueArbitrationConcern;
    concern preciseControlConcern : PreciseControlConcern;
    concern drivabilityAndResponseConcern : DrivabilityAndResponseConcern;

    #GoalConflict dependency arbitrationConflictsDrivability
        from TorqueArbitrationConcern
        to DrivabilityAndResponseConcern {
        doc
        /* Brake override limits output at once and causes jerk. Safety wins; the jerk criterion is waived during the intervention. */
    }
    #GoalConflict dependency hardwareFailureConflictsDrivability
        from HardwareFailureConcern
        to DrivabilityAndResponseConcern {
        doc
        /* A fault reaction cuts air and fuel at once and causes jerk. Safety wins once a critical fault is confirmed. */
    }
    #GoalContribution dependency preciseControlSupportsDrivability
        from PreciseControlConcern
        to DrivabilityAndResponseConcern {
        doc
        /* Less steady-state error and overshoot also reduce jerk during acceleration and deceleration. */
    }
} // package EgasArch_Needs_Concern

    private import ScalarValues::*;

    attribute def SecRating {
        attribute severity : SeverityClass;
        attribute exposure : ExposureClass;
        attribute controllability : ControllabilityClass;
    }

    item def HazardousEvent {
        doc
        /* One HARA row. rating is left out when EGAS excludes the hazardous event from consideration. */
        attribute safetyGoalId : SafetyGoalId;
        attribute hazard : String;
        attribute asil : AsilLevel;
        attribute rating : SecRating [0..1];
        attribute exclusionReason : String [0..1];
    }

    item haraSz01 : HazardousEvent {
        attribute :>> safetyGoalId = SafetyGoalId::'SZ-01';
        attribute :>> hazard = "Unintended vehicle acceleration";
        attribute :>> asil = AsilLevel::B;
        attribute :>> rating {
            attribute :>> severity = SeverityClass::S2;
            attribute :>> exposure = ExposureClass::E4;
            attribute :>> controllability = ControllabilityClass::C2;
            @SourceRef {
                document = "ISO 26262-3:2018";
                itemId = "Table 4";
            }
            @ValueStatus {
                status = ValueStatusKind::pilotAssumed;
                rationale =
                    "EGAS gives no S/E/C; S2, E4, C2 are chosen so that ISO 26262-3 Table 4 yields the EGAS ASIL B. Pending user approval.";
            }
        }
        @SourceRef {
            document = "EGAS v6.0";
            itemId = "EGAS-e83";
        }
        @ValueStatus {
            status = ValueStatusKind::'standard';
            rationale = "The safety goal, hazardous event, and ASIL are given by EGAS.";
        }
    }

    item haraSz02 : HazardousEvent {
        attribute :>> safetyGoalId = SafetyGoalId::'SZ-02';
        attribute :>> hazard = "Missing vehicle acceleration";
        attribute :>> asil = AsilLevel::QM;
        attribute :>> exclusionReason =
            "Controllable by the driver, so excluded from consideration.";
        @SourceRef {
            document = "EGAS v6.0";
            itemId = ("EGAS-e84", "EGAS-e88");
        }
        @ValueStatus {
            status = ValueStatusKind::'standard';
            rationale = "ASIL QM and the exclusion are given by EGAS; S/E/C are not evaluated.";
        }
    }

    item haraSz03 : HazardousEvent {
        attribute :>> safetyGoalId = SafetyGoalId::'SZ-03';
        attribute :>> hazard = "Unintended vehicle deceleration";
        attribute :>> asil = AsilLevel::QM;
        attribute :>> exclusionReason =
            "Controllable by the driver, so excluded from consideration.";
        @SourceRef {
            document = "EGAS v6.0";
            itemId = ("EGAS-e85", "EGAS-e88");
        }
        @ValueStatus {
            status = ValueStatusKind::'standard';
            rationale = "ASIL QM and the exclusion are given by EGAS; S/E/C are not evaluated.";
        }
    }

    item haraSz04 : HazardousEvent {
        attribute :>> safetyGoalId = SafetyGoalId::'SZ-04';
        attribute :>> hazard = "Missing vehicle deceleration";
        attribute :>> asil = AsilLevel::QM;
        attribute :>> exclusionReason =
            "Controllable by the driver, so excluded from consideration.";
        @SourceRef {
            document = "EGAS v6.0";
            itemId = ("EGAS-e86", "EGAS-e88");
        }
        @ValueStatus {
            status = ValueStatusKind::'standard';
            rationale = "ASIL QM and the exclusion are given by EGAS; S/E/C are not evaluated.";
        }
    }

    requirement def SafetyGoal {
        subject egasSystem : EgasSystem;
        attribute ftti : DurationQuantity;
        attribute asil : AsilLevel;
        item hazardousEvent : HazardousEvent;
        ref state safeState : SafeState [1..*];
    }

    requirement <sg01> UnintendedAccelerationGoal : SafetyGoal {
        doc
        /* Prevent unintended acceleration of the vehicle (EGAS safety goal SZ-01); ASIL and FTTI come from Catalog_SafetyTargets. */
        frame concern unintendedAcceleration
            : EgasArch_Needs_Concern::UnintendedAccelerationConcern;

        subject :>> egasSystem : EgasSystem;
        :>> ftti : DurationQuantity = EgasArch_Common_Catalog_SafetyTargets::sg01Ftti;
        :>> hazardousEvent : HazardousEvent = haraSz01;
        :>> asil : AsilLevel = hazardousEvent.asil;
        :>> safeState = (
            egasSystem.behavior.safeState.torqueLimited, egasSystem.behavior.safeState.fuelCutOff,
            egasSystem.behavior.safeState.fullCutOff
        );

        @SourceRef {
            document = "EGAS v6.0";
            itemId = "EGAS-e87";
        }
        @ValueStatus {
            status = ValueStatusKind::pilotAssumed;
            rationale =
                "Covers safeState only; ASIL and FTTI status are in the catalog. EGAS requires a safe state but does not name the stage. 
                TorqueLimited is a pilot choice and may not remove the hazard if the torque path itself fails; T12 and T19 must confirm. Pending user approval.";
        }
    }

    // Derivation traceability: Safety Concerns -> Safety Goal (SG-01)
    #derivation connection sg01FromUnintendedAcceleration {
        end #original concern ::> unintendedAccelerationConcern;
        end #derive requirement ::> sg01;
    }

    #derivation connection sg01FromHardwareFailure {
        end #original concern ::> hardwareFailureConcern;
        end #derive requirement ::> sg01;
    }

    #derivation connection sg01FromTorqueArbitration {
        end #original concern ::> torqueArbitrationConcern;
        end #derive requirement ::> sg01;
    }
} // package EgasArch_Safety_Goal

package EgasArch_Ctx_Context {
    private import EgasArch_Common_Signals::*;
    private import EgasArch_Needs_Stakeholder::Driver;
    private import EgasArch_Beh_Behavior::*;

    // Port Definitions (EGAS-System Boundary & Internal Interfaces)

    port def TorqueMonitorPort {
        doc
        /* ECU 내부 감시 포트. 감시 토크 값을 노출하며 EgasContext의 외부 connect 대상이 아니다. */
    }
    port def DriverPedalActuationPort {
        doc
        /* Driver physical actuation -> AccelerationPedal (mechanical/human input) */
        in item actuation : DriverPedalActuation;
    }
    port def AccelerationInputPort {
        doc
        /* Internal EGAS: AccelerationPedal -> EngineControlUnit (PVS1, PVS2) */

        in item pedalPosition : AccelerationPedalPosition;
    }

    port def ThrottleControlPort {
        doc
        /* ECU와 스로틀 밸브 간 제어 명령 및 이중 위치 센서(TPS1/TPS2) 피드백 포트. 
         * (추후 Common_Signals에 ThrottleCommand / ThrottlePosition 추가 시 item 바인딩)
         * Internal EGAS: EngineControlUnit <-> ThrottleValve (actuator drive & TPS1/TPS2 feedback)
         */
        out item command : ThrottleControlCommand;
        in item feedback : ThrottlePositionFeedback;
    }

    port def CruiseControlPort {
        doc
        /* 크루즈 컨트롤 레버(옵션) 입력 포트 (EGAS v6.0 Fig 1).
         * External: Cruise Control Lever (opt) -> EngineControlUnit
         */
        in item ccRequest : CruiseControlRequest;
    }

    port def BrakeInputPort {
        doc
        /* External: Brake -> EngineControlUnit */
        in item brakePosition : BrakePedalPosition;
    }

    port def SensorInputPort {
        doc
        /* External: Engine / Additional Torque-Affecting Sensors -> EngineControlUnit */
        in item engineSensor : EngineSensorData;
    }

    port def TorqueInterventionPort {
        doc
        /* External: Add. ECU (TCU, ESP, AC, etc.) -> EngineControlUnit */
        in item torqueRequest : TorqueInterventionRequest;
    }

    port def AirIntakePort {
        doc
        /* External: ThrottleValve -> Engine (air path) */
        out item airFlow : IntakeAirFlow;
    }

    port def FuelCommandPort {
        doc
        /* External: EngineControlUnit -> Fuel Injection / Ignition System */
        out item fuelCommand : FuelIgnitionCommand;
    }

    port def DashboardPort {
        doc
        /* External: EngineControlUnit -> Dashboard HMI (EGAS_e52 optical driver warning) */
        out item warning : DashboardWarning;
    }

    port def DriverBrakeActuationPort {
        doc
        /* 운전자가 브레이크 페달을 물리적으로 조작하는 인터페이스. */
    }

    // Top-level Vehicle Safe States (Single Source of Truth; Beh_Behavior imports these)
    state def OffState;
    state def DrivingState;
    state def SafeState;
    state def TorqueLimited :> SafeState; // 1단계: 제어 가능한 출력/토크 제한 주행
    state def FuelCutOff :> SafeState; // 2단계: 추진력 차단 (연료 분사 차단 ICO 등)
    state def FullCutOff :> SafeState; // 3단계: 출력단 완전 차단 / 엔진 정지 / 시동 차단

    // EGAS-System Internal Subsystems (EGAS v6.0 Chapter 7, EGAS_e74~e78, Fig 1)

    part def AccelerationPedal {
        doc
        /* EGAS 내부 서브시스템 (EGAS_e75, SReq-01): 운전자의 가속 조작을 이중 채널(PVS1/PVS2) 신호로 ECU에 전달한다. */
        port driverInput : DriverPedalActuationPort;
        port pedalOutput : ~AccelerationInputPort;
    }

    part def EngineControlUnit {
        doc
        /* EGAS 내부 서브시스템 (EGAS_e76, SReq-03~07): 3-Level 모니터링(L1/L2/L3) 및 토크 제어를 수행하는 엔진 제어기. */
        port torqueMonitor : TorqueMonitorPort;
        port accelerationInput : AccelerationInputPort;
        port throttleControl : ThrottleControlPort;
        port cruiseControlInput : CruiseControlPort;
        port brakeInput : BrakeInputPort;
        port sensorInput : SensorInputPort;
        port torqueIntervention : TorqueInterventionPort;
        port fuelCommand : FuelCommandPort;
        port dashboardOutput : DashboardPort;
    }

    part def ThrottleValve {
        doc
        /* EGAS 내부 서브시스템 (EGAS_e77, SReq-02): ECU 명령으로 흡기 공기량을 제어하고 이중 위치 센서(TPS1/TPS2)를 피드백한다. */
        port throttleControl : ~ThrottleControlPort;
        port airIntake : AirIntakePort;
    }

    // System of Interest: EGAS-System (EGAS v6.0 Fig 1 dashed boundary)
    part def EgasSystem {
        doc
        /* EGAS v6.0 Fig 1 기준 시스템 경계: AccelerationPedal, EngineControlUnit, ThrottleValve 3개 서브시스템으로 구성된다. */

        // System Boundary Ports
        port driverPedalInput : DriverPedalActuationPort;
        port cruiseControlInput : CruiseControlPort;
        port brakeInput : BrakeInputPort;
        port sensorInput : SensorInputPort;
        port torqueIntervention : TorqueInterventionPort;
        port airIntake : AirIntakePort;
        port fuelCommand : FuelCommandPort;
        port dashboardOutput : DashboardPort;

        // Internal Subsystems (EGAS_e75~e77)
        part accelerationPedal : AccelerationPedal;
        part ecu : EngineControlUnit;
        part throttleValve : ThrottleValve;

        // Boundary Port Delegations
        bind driverPedalInput = accelerationPedal.driverInput;
        bind cruiseControlInput = ecu.cruiseControlInput;
        bind brakeInput = ecu.brakeInput;
        bind sensorInput = ecu.sensorInput;
        bind torqueIntervention = ecu.torqueIntervention;
        bind airIntake = throttleValve.airIntake;
        bind fuelCommand = ecu.fuelCommand;
        bind dashboardOutput = ecu.dashboardOutput;

        // Internal Subsystem Interconnections (Fig 1)
        connect accelerationPedal.pedalOutput to ecu.accelerationInput;
        connect ecu.throttleControl to throttleValve.throttleControl;

        // Observable System Fault / Mode Events
        event occurrence unintendedAcceleration : UnintendedAccelerationEvent;
        event occurrence sensorOrActuatorFault : SensorOrActuatorFaultEvent;
        event occurrence externalRequestFault : ExternalRequestFaultEvent;
        event occurrence controllerFault : ControllerFaultEvent;
        event occurrence interventionReleased : InterventionReleasedEvent;

        // Boundary Message Event Occurrences (Used by EgasArch_Scn_Scenario messages)
        event occurrence pedalActuationReceived;
        event occurrence ccRequestReceived;
        event occurrence brakePositionReceived;
        event occurrence externalRequestReceived;
        event occurrence engineSensorReceived;
        event occurrence airFlowSent;
        event occurrence fuelCommandSent;
        event occurrence warningSent;

        exhibit state behavior : EgasBehavior;
    }

    // External Actors / Systems (Outside EGAS-System Boundary, Fig 1)
    part def CruiseControlLever {
        doc
        /* 크루즈 컨트롤 조작 레버 (옵션, EGAS v6.0 Fig 1). */
        port ccOutput : ~CruiseControlPort;
        event occurrence ccRequestSent;
    }

    part def Brake {
        doc
        /* 제동 의지를 ECU에 전달하는 브레이크 시스템/스위치 (EGAS v6.0 Fig 1). */
        port driverInput : DriverBrakeActuationPort;
        port brakeOutput : ~BrakeInputPort;
        event occurrence brakeActuationReceived;
        event occurrence brakePositionSent;
    }

    part def AddEcu {
        doc
        /* 외부 제어기(TCU, ESP, AC 등)로 ECU에 토크 개입을 요청한다 (EGAS v6.0 Fig 1). */
        port requestOutput : ~TorqueInterventionPort;
        event occurrence externalRequestSent;
    }

    part def Engine {
        doc
        /* 추가 토크 영향 센서(엔진 회전수, 부하 등) 정보를 ECU에 전달하고 스로틀 밸브로부터 흡기 공기를 받는 물리적 엔진
         * (EGAS v6.0 Fig 1: Add. Sensors/Components + Combustion Engine).
         */
        port sensorOutput : ~SensorInputPort;
        port airInput : ~AirIntakePort;
        event occurrence engineSensorSent;
        event occurrence airFlowReceived;
    }

    part def FuelInjectionIgnitionSystem {
        doc
        /* ECU의 연료 분사·점화 명령을 받는 외부 액추에이터 시스템 (EGAS v6.0 Fig 1). */
        port commandInput : ~FuelCommandPort;
        event occurrence fuelCommandReceived;
    }

    part def DashboardHMI {
        doc
        /* ECU의 경고 상태를 운전자에게 광학적으로 표시한다 (EGAS_e52). */
        port warningInput : ~DashboardPort;
        event occurrence warningReceived;
    }

    // System Context (EGAS-System + External Environment)
    part def EgasContext {
        doc
        /* System context aligned with EGAS v6.0 Fig 1: EGAS-System boundary and its external actors. */
        part egas : EgasSystem;
        part driver : Driver;
        part ccLever : CruiseControlLever [0..1];
        part brake : Brake;
        part externalEcu : AddEcu;
        part engine : Engine;
        part fuelSystem : FuelInjectionIgnitionSystem;
        part dashboard : DashboardHMI;

        connect driver.pedalActuation to egas.driverPedalInput;
        connect driver.brakeActuation to brake.driverInput;
        connect brake.brakeOutput to egas.brakeInput;
        connect ccLever.ccOutput to egas.cruiseControlInput;
        connect externalEcu.requestOutput to egas.torqueIntervention;
        connect engine.sensorOutput to egas.sensorInput;
        connect egas.airIntake to engine.airInput;
        connect egas.fuelCommand to fuelSystem.commandInput;
        connect egas.dashboardOutput to dashboard.warningInput;
    }
} // package EgasArch_Ctx_Context

package EgasArch_Shr_Requirement {

    // Nominal & Process Stakeholder Requirements (Functional Safety Requirements are separated in EgasArch_Shr_Fsr)

    requirement def StakeholderRequirement;

    requirement <'Shr-01'> SteadyStatePrecisionRequirement : StakeholderRequirement {
        doc
        /* The delivered torque shall track the requested torque in steady state with a small error 
         * and a small overshoot (legacy SHR-02, SHR-02-1).
         */
        subject egas : EgasSystem;

        @GoalData {
            kind = GoalKind::soft;
        }
        @SourceRef {
            document = "Legacy SHR-02-1 (Tier 1 supplier interview)";
        }
        @ValueStatus {
            status = ValueStatusKind::pilotAssumed;
            rationale =
                "The limit is the pilotAssumed catalog entry steadyStateTrackingLimit; EGAS gives none. The overshoot limit is not set.";
        }
    }

    requirement <'Shr-02'> TorqueInterventionRequirement : StakeholderRequirement {
        doc
        /* When another ECU requests a nominal torque reduction (e.g., chassis stability control),
         * the system shall reduce air, fuel, and ignition commands within the intervention response budget 
         * and return to driver-request control once released (legacy SHR-04, SHR-04-1, SHR-04-2, SHR-04-4). 
         * Safety protection of external torque-increasing requests is covered by FSR-04.
         */

        subject egas : EgasSystem;

        @GoalData {
            kind = GoalKind::hard;
            refinement = GoalRefinementKind::AND;
        }
        @SourceRef {
            document = "Legacy SHR-04 (chassis control meeting minutes)";
        }
        @ValueStatus {
            status = ValueStatusKind::pilotAssumed;
            rationale =
                "The budget is the pilotAssumed catalog entry interventionResponseBudget; EGAS gives none. The throttle-stuck fallback (legacy SHR-04-3) is demoted.";
        }
    }

    requirement <'Shr-03'> PedalResponseRequirement : StakeholderRequirement {
        doc
        /* After a pedal input the system shall output the air and the fuel commands within the pedal response budget; 
         * safety reactions defined in EgasArch_Shr_Fsr take priority (legacy SHR-08).
         */

        subject egas : EgasSystem;

        @GoalData {
            kind = GoalKind::soft;
        }
        @SourceRef {
            document = "Legacy SHR-08 (vehicle dynamics and ergonomics guidelines)";
        }
        @ValueStatus {
            status = ValueStatusKind::pilotAssumed;
            rationale =
                "The budget is the pilotAssumed catalog entry pedalResponseBudget; EGAS gives none.";
        }
    }

    requirement <'Shr-04'> FunctionalSafetyComplianceRequirement : StakeholderRequirement {
        doc
        /* The development of the system shall follow ISO 26262 for the ASIL of the safety goal (legacy SHR-P01). */
        subject egas : EgasSystem;

        frame concern functionalSafety : FunctionalSafetyConcern;

        @GoalData {
            kind = GoalKind::hard;
        }
        @SourceRef {
            document = "EGAS v6.0";
            itemId = "EGAS-e90";
        }
        @ValueStatus {
            status = ValueStatusKind::'standard';
            rationale =
                "EGAS requires implementation to the safety goal ASIL (ASIL B); the legacy ASIL-D text is not carried.";
        }
    }

    // Derivation traceability: Concerns -> Stakeholder Requirements
    #derivation connection precisionShrFromConcern {
        end #original requirement ::> preciseControlConcern;
        end #derive requirement ::> SteadyStatePrecisionRequirement;
    }

    #derivation connection interventionShrFromConcern {
        end #original requirement ::> torqueArbitrationConcern;
        end #derive requirement ::> TorqueInterventionRequirement;
    }

    #derivation connection pedalResponseShrFromConcern {
        end #original requirement ::> drivabilityAndResponseConcern;
        end #derive requirement ::> PedalResponseRequirement;
    }

    #derivation connection complianceShrFromConcern {
        end #original requirement ::> functionalSafetyConcern;
        end #derive requirement ::> FunctionalSafetyComplianceRequirement;
    }
} // package EgasArch_Shr_Requirement

    // -------------------------------------------------------------------------
    // UC_Common: Safe State Transition & Driver Warning (Included Utility UC)
    // -------------------------------------------------------------------------
    use case <UC_Common> transitionToSafeState : SafetyUseCase {
        doc
        /* Reusable black-box safety capability: limit or cut off propulsion output
         * and optically warn the driver upon a confirmed fault (EGAS_e52, EGAS_e87).
         * ASIL and FTTI are governed by the parent SafetyUseCase that includes this UC.
         */
        subject :>> egasSystem : EgasSystem;
        actor engine : Engine;
        actor fuelSystem : FuelInjectionIgnitionSystem;
        actor dashboard : DashboardHMI;

        // Parameters passed from parent UC (UC2 / UC3) on include
        in item ucEngineSensor : EngineSensorData;
        in attribute ucCutOffRequired : Boolean;
        out item ucWarning : DashboardWarning;

        assert constraint inv_systemOperational : Invariant {
            doc
            /* The EGAS system power supply and safety reaction paths remain operational throughout the transition. */
            isInState(egasSystem.behavior)
        }

        objective reachSafeStateObjective {
            doc
            /* Limit or cut off propulsion output and warn the driver optically. */
        }

        // 1. 시스템 내부 안전 반응 액션
        action limitPower : LimitPropulsionPower {
            in :>> engineSensor = ucEngineSensor;
        }
        action cutOff : CutOffPropulsion;
        action warn : WarnDriver;

        bind warn.warning = ucWarning;

        // 2. 경계 출력 송신 (Send Actions — 분기별 송신 포트 + 수신 Actor 명시)
        action sendLimitedAir send limitPower.airFlow via egasSystem.airIntake to engine;
        action sendLimitedFuel send limitPower.fuelCommand via egasSystem.fuelCommand to fuelSystem;

        action sendCutOffAir send cutOff.airFlow via egasSystem.airIntake to engine;
        action sendCutOffFuel send cutOff.fuelCommand via egasSystem.fuelCommand to fuelSystem;

        action sendDriverWarning send warn.warning via egasSystem.dashboardOutput to dashboard;

        // 3. Send Action -> Actor Event Occurrence 간 Message 연결
        message limitedAirOutput of IntakeAirFlow from sendLimitedAir to engine.airFlowReceived;
        message limitedFuelOutput of FuelIgnitionCommand
            from sendLimitedFuel to fuelSystem.fuelCommandReceived;

        message cutOffAirOutput of IntakeAirFlow from sendCutOffAir to engine.airFlowReceived;
        message cutOffFuelOutput of FuelIgnitionCommand
            from sendCutOffFuel to fuelSystem.fuelCommandReceived;

        message driverWarningOutput of DashboardWarning
            from sendDriverWarning to dashboard.warningReceived;

        // 4. 제어 흐름 (Control Flow)
        decide selectStage;
        fork sendLimitedOutputs;
        join limitedOutputsSent;
        fork sendCutOffOutputs;
        join cutOffOutputsSent;
        merge stageDone;

        first start then selectStage;

        // Branch 1: Stage 1 (TorqueLimited)
        first selectStage if not ucCutOffRequired then limitPower;
        first limitPower then sendLimitedOutputs;
        first sendLimitedOutputs then sendLimitedAir;
        first sendLimitedOutputs then sendLimitedFuel;
        first sendLimitedAir then limitedOutputsSent;
        first sendLimitedFuel then limitedOutputsSent;
        first limitedOutputsSent then stageDone;

        // Branch 2: Stage 2/3 (FuelCutOff / FullCutOff)
        first selectStage if ucCutOffRequired then cutOff;
        first cutOff then sendCutOffOutputs;
        first sendCutOffOutputs then sendCutOffAir;
        first sendCutOffOutputs then sendCutOffFuel;
        first sendCutOffAir then cutOffOutputsSent;
        first sendCutOffFuel then cutOffOutputsSent;
        first cutOffOutputsSent then stageDone;

        first stageDone then warn;
        first warn then sendDriverWarning;
        first sendDriverWarning then done;
    }

    // -------------------------------------------------------------------------
    // UC1: Deliver Driver Torque (Nominal Capability)
    // -------------------------------------------------------------------------
    use case <UC1> deliverDriverTorque : NominalUseCase {
        doc
        /* Nominal capability: respond to the driver's accelerator pedal input by delivering
         * the requested engine torque accurately and predictably.
         */

        subject :>> egasSystem : EgasSystem;
        actor driver : Driver;
        actor engine : Engine;
        actor fuelSystem : FuelInjectionIgnitionSystem;

        objective deliverDriverTorqueObjective {
            doc
            /* Track driver torque demand in steady state and respond within the pedal response budget. */
            assume constraint pre_drivingMode : Precondition {
                isInState(egasSystem.behavior.normalDriving.driverDemand)
            }
            require constraint post_nominalDelivery : Postcondition {
                DurationOf(deliver) <= pedalResponseBudget and
                deliver.trackingError <= steadyStateTrackingLimit
            }
        }

        // 1. 경계 입력 수신 (Accept Actions — 수신 포트 명시)
        action receivePedal accept pedal : DriverPedalActuation via egasSystem.driverPedalInput;
        action receiveEngineSensor accept sensor : EngineSensorData via egasSystem.sensorInput;

        // 2. 시스템 내부 기능 수행 (EgasArch_Beh_Behavior::DeliverRequestedPropulsion)
        action deliver : DeliverRequestedPropulsion;

        // 3. 경계 출력 송신 (Send Actions — 송신 포트 + 수신 Actor 명시)
        action sendAirFlow send deliver.airFlow via egasSystem.airIntake to engine;
        action sendFuelCommand send deliver.fuelCommand via egasSystem.fuelCommand to fuelSystem;

        // 4. Actor Event Occurrence <-> Accept/Send Action 간 Message 연결 (Sequence View 연계)
        message pedalInput of DriverPedalActuation from driver.pedalActuationSent to receivePedal;
        message engineSensorInput of EngineSensorData
            from engine.engineSensorSent to receiveEngineSensor;
        message airFlowOutput of IntakeAirFlow from sendAirFlow to engine.airFlowReceived;
        message fuelCommandOutput of FuelIgnitionCommand
            from sendFuelCommand to fuelSystem.fuelCommandReceived;

        // 5. 데이터 흐름 (Accept 출력 -> DeliverRequestedPropulsion 입력)
        flow receivePedal.pedal to deliver.driverActuation;
        flow receiveEngineSensor.sensor to deliver.engineSensor;

        // 6. 제어 흐름 (Control Flow)
        merge loopEntry;
        fork receiveInputs;
        join inputsReady;
        fork sendOutputs;
        join outputsSent;
        decide continueDriving;

        first start then loopEntry;
        first loopEntry then receiveInputs;

        first receiveInputs then receivePedal;
        first receiveInputs then receiveEngineSensor;
        first receivePedal then inputsReady;
        first receiveEngineSensor then inputsReady;

        first inputsReady then deliver;
        first deliver then sendOutputs;

        first sendOutputs then sendAirFlow;
        first sendOutputs then sendFuelCommand;
        first sendAirFlow then outputsSent;
        first sendFuelCommand then outputsSent;

        first outputsSent then continueDriving;
        first continueDriving
            if isInState(egasSystem.behavior.normalDriving.driverDemand) then loopEntry;
        first continueDriving
            if not isInState(egasSystem.behavior.normalDriving.driverDemand) then done;

        dependency uc1RefinesShrs
            from deliverDriverTorque
            to SteadyStatePrecisionRequirement, PedalResponseRequirement;
    }

    // -------------------------------------------------------------------------
    // UC2: Arbitrate Conflicting Torque Requests (Intervention & Brake Override)
    // -------------------------------------------------------------------------
    use case <UC2> arbitrateTorqueRequests : SafetyUseCase {
        doc
        /* Arbitration capability: resolve conflicting torque requests from the driver,
         * cruise control, service brake, and external ECUs, operating across normalDriving
         * (externalIntervention, brakeOverride) and safeState on faulty external requests
         * (SHR-04, FSR-04, FSR-08).
         */
        subject :>> egasSystem : EgasSystem;
        actor driver : Driver;
        actor ccLever : CruiseControlLever;
        actor brake : Brake;
        actor externalEcu : AddEcu;
        actor engine : Engine;
        actor fuelSystem : FuelInjectionIgnitionSystem;
        actor dashboard : DashboardHMI;

        :>> asil = sg01Asil;
        :>> ftti = sg01Ftti;

        assert constraint inv_systemOperational : Invariant {
            doc
            /* The EGAS system remains operational throughout arbitration or fault transition. */
            isInState(egasSystem.behavior)
        }

        // 1. 경계 입력 수신 (Accept Actions — 수신 포트 명시)
        action receivePedal accept actuation : DriverPedalActuation via egasSystem.driverPedalInput;
        action receiveCcRequest accept cc : CruiseControlRequest via egasSystem.cruiseControlInput;
        action receiveBrake accept brakePos : BrakePedalPosition via egasSystem.brakeInput;
        action receiveExternalRequest
            accept extReq : TorqueInterventionRequest via egasSystem.torqueIntervention;
        action receiveEngineSensor accept sensorData : EngineSensorData via egasSystem.sensorInput;

        // 2. 시스템 내부 중재/안전 액션 및 Included UC
        action rejectExternal : RejectFaultyExternalRequest;
        action overrideOnBrake : ApplyBrakeOverride;
        action reduceByExternalEcu : ApplyExternalTorqueReduction;

        include use case enterSafeState :> transitionToSafeState {
            subject :>> egasSystem : EgasSystem;
            in :>> ucEngineSensor = receiveEngineSensor.sensorData;
            in :>> ucCutOffRequired = false;
        }

        // 3. 경계 출력 송신 (Send Actions — 분기별 송신 포트 + 수신 Actor 명시)
        action sendOverrideAir send overrideOnBrake.airFlow via egasSystem.airIntake to engine;
        action sendOverrideFuel
            send overrideOnBrake.fuelCommand via egasSystem.fuelCommand to fuelSystem;

        action sendReducedAir send reduceByExternalEcu.airFlow via egasSystem.airIntake to engine;
        action sendReducedFuel
            send reduceByExternalEcu.fuelCommand via egasSystem.fuelCommand to fuelSystem;

        // 4. Actor Event Occurrence <-> Accept/Send Action 간 Message 연결
        message pedalInput of DriverPedalActuation from driver.pedalActuationSent to receivePedal;
        message ccRequestInput of CruiseControlRequest
            from ccLever.ccRequestSent to receiveCcRequest;
        message pressBrakePedal from driver.brakeActuationSent to brake.brakeActuationReceived;
        message brakeInput of BrakePedalPosition from brake.brakePositionSent to receiveBrake;
        message externalRequestInput of TorqueInterventionRequest
            from externalEcu.externalRequestSent to receiveExternalRequest;
        message engineSensorInput of EngineSensorData
            from engine.engineSensorSent to receiveEngineSensor;

        message overrideAirOutput of IntakeAirFlow from sendOverrideAir to engine.airFlowReceived;
        message overrideFuelOutput of FuelIgnitionCommand
            from sendOverrideFuel to fuelSystem.fuelCommandReceived;

        message reducedAirOutput of IntakeAirFlow from sendReducedAir to engine.airFlowReceived;
        message reducedFuelOutput of FuelIgnitionCommand
            from sendReducedFuel to fuelSystem.fuelCommandReceived;

        // 5. 데이터 흐름 (Accept 출력 -> 중재 액션 입력)
        flow receiveExternalRequest.extReq to rejectExternal.externalRequest;

        flow receivePedal.actuation to overrideOnBrake.driverActuation;
        flow receiveCcRequest.cc to overrideOnBrake.ccRequest;
        flow receiveBrake.brakePos to overrideOnBrake.brakePosition;
        flow receiveEngineSensor.sensorData to overrideOnBrake.engineSensor;

        flow receiveExternalRequest.extReq to reduceByExternalEcu.externalRequest;
        flow receiveEngineSensor.sensorData to reduceByExternalEcu.engineSensor;

        objective arbitrateTorqueObjective {
            doc
            /* Enforce brake override, reject faulty external torque-increasing requests within FTTI, and apply valid torque reductions within budget. */
            assume constraint pre_activeArbitration : Precondition {
                isInState(egasSystem.behavior.normalDriving) and
                (receiveBrake.brakePos.stopLightSwitchActive or
                    receiveCcRequest.cc.active or
                    isInState(egasSystem.behavior.safeState))
            }
            require constraint post_onExternalFault : Postcondition {
                isInState(egasSystem.behavior.safeState) implies
                (isInState(egasSystem.behavior.safeState.torqueLimited) and
                    DurationOf(enterSafeState) <= ftti and
                    enterSafeState.ucWarning.warningActive)
            }
            require constraint post_onBrakeOverride : Postcondition {
                (isInState(egasSystem.behavior.normalDriving) and
                    receiveBrake.brakePos.stopLightSwitchActive) implies
                (isInState(egasSystem.behavior.normalDriving.brakeOverride) and
                    DurationOf(overrideOnBrake) <= interventionResponseBudget)
            }
            require constraint post_onExternalReduction : Postcondition {
                (isInState(egasSystem.behavior.normalDriving) and
                    not receiveBrake.brakePos.stopLightSwitchActive) implies
                (isInState(egasSystem.behavior.normalDriving.externalIntervention) and
                    DurationOf(reduceByExternalEcu) <= interventionResponseBudget)
            }
        }

        // 6. 제어 흐름 (Control Flow)
        fork receiveInputs;
        join inputsReady;
        decide checkArbitrationCase;
        fork sendOverrideOutputs;
        join overrideOutputsSent;
        fork sendReducedOutputs;
        join reducedOutputsSent;
        merge arbitrationDone;

        first start then receiveInputs;
        first receiveInputs then receivePedal;
        first receiveInputs then receiveCcRequest;
        first receiveInputs then receiveBrake;
        first receiveInputs then receiveExternalRequest;
        first receiveInputs then receiveEngineSensor;

        first receivePedal then inputsReady;
        first receiveCcRequest then inputsReady;
        first receiveBrake then inputsReady;
        first receiveExternalRequest then inputsReady;
        first receiveEngineSensor then inputsReady;

        first inputsReady then checkArbitrationCase;

        // Branch 1: Faulty external request -> reject & enterSafeState
        first checkArbitrationCase if isInState(egasSystem.behavior.safeState) then rejectExternal;
        first rejectExternal then enterSafeState;
        first enterSafeState then arbitrationDone;

        // Branch 2: Brake override -> send override air & fuel
        first checkArbitrationCase
            if (
                isInState(egasSystem.behavior.normalDriving) and
                receiveBrake.brakePos.stopLightSwitchActive
            )
                then overrideOnBrake;
        first overrideOnBrake then sendOverrideOutputs;
        first sendOverrideOutputs then sendOverrideAir;
        first sendOverrideOutputs then sendOverrideFuel;
        first sendOverrideAir then overrideOutputsSent;
        first sendOverrideFuel then overrideOutputsSent;
        first overrideOutputsSent then arbitrationDone;

        // Branch 3: Valid external ECU reduction -> send reduced air & fuel
        first checkArbitrationCase
            if (
                isInState(egasSystem.behavior.normalDriving) and
                not receiveBrake.brakePos.stopLightSwitchActive
            )
                then reduceByExternalEcu;
        first reduceByExternalEcu then sendReducedOutputs;
        first sendReducedOutputs then sendReducedAir;
        first sendReducedOutputs then sendReducedFuel;
        first sendReducedAir then reducedOutputsSent;
        first sendReducedFuel then reducedOutputsSent;
        first reducedOutputsSent then arbitrationDone;

        first arbitrationDone then done;

        dependency uc2RefinesReqs
            from arbitrateTorqueRequests
            to
                TorqueInterventionRequirement,
                fsr04_ExternalTorqueRequestProtection,
                fsr08_BrakeOverride;
    }

        actor engine : Engine;
        actor fuelSystem : FuelInjectionIgnitionSystem;
        actor dashboard : DashboardHMI;

        :>> asil = sg01Asil;
        :>> ftti = sg01Ftti;

        assert constraint inv_systemOperational : Invariant {
            doc
            /* The EGAS system remains operational with independent monitoring hardware active. */
            isInState(egasSystem.behavior)
        }

        // 1. 경계 입력 수신 (Accept Actions — 수신 포트 명시)
        action receivePedal accept actuation : DriverPedalActuation via egasSystem.driverPedalInput;
        action receiveEngineSensor accept sensorData : EngineSensorData via egasSystem.sensorInput;

        // 2. 시스템 내부 기능 수행 및 Included Safe State UC
        action deliver : DeliverRequestedPropulsion;

        include use case enterSafeState :> transitionToSafeState {
            subject :>> egasSystem : EgasSystem;
            in :>> ucEngineSensor = receiveEngineSensor.sensorData;
            in :>> ucCutOffRequired =
                isInState(egasSystem.behavior.safeState.fuelCutOff) or
                isInState(egasSystem.behavior.safeState.fullCutOff);
        }

        // 3. 경계 출력 송신 (Normal Path — 송신 포트 + 수신 Actor 명시)
        action sendAirFlow send deliver.airFlow via egasSystem.airIntake to engine;
        action sendFuelCommand send deliver.fuelCommand via egasSystem.fuelCommand to fuelSystem;

        // 4. Actor Event Occurrence <-> Accept/Send Action 간 Message 연결
        message pedalInput of DriverPedalActuation from driver.pedalActuationSent to receivePedal;
        message engineSensorInput of EngineSensorData
            from engine.engineSensorSent to receiveEngineSensor;
        message airFlowOutput of IntakeAirFlow from sendAirFlow to engine.airFlowReceived;
        message fuelCommandOutput of FuelIgnitionCommand
            from sendFuelCommand to fuelSystem.fuelCommandReceived;

        // 5. 데이터 흐름 (Accept 출력 -> DeliverRequestedPropulsion 입력)
        flow receivePedal.actuation to deliver.driverActuation;
        flow receiveEngineSensor.sensorData to deliver.engineSensor;

        objective protectAgainstUnintendedAccelerationObjective {
            doc
            /* Prevent unintended acceleration upon confirmed fault by transitioning to safeState within FTTI and warning the driver. */
            assume constraint pre_monitoringActive : Precondition {
                isInState(egasSystem.behavior.normalDriving)
            }
            require constraint post_safeStateWithinFtti : Postcondition {
                isInState(egasSystem.behavior.safeState) and
                DurationOf(enterSafeState) <= ftti and
                enterSafeState.ucWarning.warningActive
            }
        }

        // 6. 제어 흐름 (Control Flow)
        merge monitorLoop;
        fork receiveInputs;
        join inputsReady;
        decide faultCheck;
        fork sendNormalOutputs;
        join normalOutputsSent;

        first start then monitorLoop;
        first monitorLoop then receiveInputs;

        first receiveInputs then receivePedal;
        first receiveInputs then receiveEngineSensor;
        first receivePedal then inputsReady;
        first receiveEngineSensor then inputsReady;

        first inputsReady then deliver;
        first deliver then faultCheck;

        // Normal path: send air & fuel commands and continue monitoring loop
        first faultCheck if isInState(egasSystem.behavior.normalDriving) then sendNormalOutputs;
        first sendNormalOutputs then sendAirFlow;
        first sendNormalOutputs then sendFuelCommand;
        first sendAirFlow then normalOutputsSent;
        first sendFuelCommand then normalOutputsSent;
        first normalOutputsSent then monitorLoop;

        // Fault path: execute included transitionToSafeState (handles safe air/fuel & warning)
        first faultCheck if isInState(egasSystem.behavior.safeState) then enterSafeState;
        first enterSafeState then done;

        dependency uc3RefinesFsrs
            from monitorAndProtectTorque
            to
                fsr01_PedalSensorPlausibility,
                fsr02_ThrottleSensorPlausibility,
                fsr03_SensorSystemFaultDetection,
                fsr05_ActuatorErrorDetection,
                fsr06_TorqueExcessSafeState,
                fsr07_FunctionControllerMonitoring;
    }
} // package EgasArch_Scn_Scenario

package EgasArch_Beh_Behavior {
    private import ScalarValues::*;
    private import ModelingMetadata::*;
    private import EgasArch_Common_Units::PercentQuantity;
    private import EgasArch_Common_Signals::*;
    private import EgasArch_Ctx_Context::*;
    private import EgasArch_Shr_Requirement::*;
    private import EgasArch_Shr_Fsr::*;

    // System-Level Observable Events (Occurrence Definitions -> allocated to EgasSystem)
    occurrence def UnintendedAccelerationEvent {
        doc
        /* Observable onset of impermissible driving torque or unintended vehicle acceleration (EGAS SReq-06). */
    }
    occurrence def SensorOrActuatorFaultEvent {
        doc
        /* Confirmed fault of accelerator pedal, throttle valve, brake sensor, or actuator (EGAS SReq-01~03, 05). */
        attribute cutOffRequired : Boolean;
    }
    occurrence def ExternalRequestFaultEvent {
        doc
        /* Faulty, corrupted, or missing torque-increasing request from an external ECU (EGAS SReq-04). */
    }
    occurrence def ControllerFaultEvent {
        doc
        /* Confirmed integrity fault of the engine function controller (EGAS SReq-07). */
    }
    occurrence def InterventionReleasedEvent {
        doc
        /* External torque reduction request or brake override is released. */
    }

    // Externally Observable Black-Box Actions (No responseTime / safeStateReached outputs)
    action def DeliverRequestedPropulsion {
        doc
        /* Delivers air and fuel/ignition commands matching the driver's pedal actuation. */
        in driverActuation : DriverPedalActuation;
        in engineSensor : EngineSensorData;
        out airFlow : IntakeAirFlow;
        out fuelCommand : FuelIgnitionCommand;
        out trackingError : PercentQuantity;
    }

    action def ApplyBrakeOverride {
        doc
        /* Limits propulsion power to a controllable maximum while brake and propulsion demand overlap (EGAS 10.1.2.4). */
        in driverActuation : DriverPedalActuation;
        in ccRequest : CruiseControlRequest;
        in brakePosition : BrakePedalPosition;
        in engineSensor : EngineSensorData;
        out airFlow : IntakeAirFlow;
        out fuelCommand : FuelIgnitionCommand;
    }

    action def ApplyExternalTorqueReduction {
        doc
        /* Reduces air and fuel/ignition output per a valid external ECU torque-reduction request (SHR-04). */
        in externalRequest : TorqueInterventionRequest;
        in engineSensor : EngineSensorData;
        out airFlow : IntakeAirFlow;
        out fuelCommand : FuelIgnitionCommand;
    }

    action def RejectFaultyExternalRequest {
        doc
        /* Inhibits a faulty or missing external torque-increasing request (EGAS SReq-04). */
        in externalRequest : TorqueInterventionRequest;
    }

    action def LimitPropulsionPower {
        doc
        /* Stage 1 Safe State (TorqueLimited): limits vehicle propulsion output (limp-home / idle demand). */
        in engineSensor : EngineSensorData;
        out airFlow : IntakeAirFlow;
        out fuelCommand : FuelIgnitionCommand;
    }

    action def CutOffPropulsion {
        doc
        /* Stage 2/3 Safe State (FuelCutOff / FullCutOff): cuts off fuel injection (ICO) or switches off power stages. */
        out airFlow : IntakeAirFlow;
        out fuelCommand : FuelIgnitionCommand;
    }

    action def WarnDriver {
        doc
        /* Raises an optical warning on the dashboard HMI upon fault reaction (EGAS_e52). */
        out warning : DashboardWarning;
    }

    // System-Level Black-Box State Machine
    state def EgasBehavior {
        doc
        /* System-level black-box behavior of the EGAS system:
         * normalDriving (driverDemand / externalIntervention / brakeOverride)
         * and staged safeState (TorqueLimited / FuelCutOff / FullCutOff).
         */
        entry;
        then normalDriving;

        state normalDriving {
            doc
            /* Normal vehicle operation: serves driver demand and handles nominal external intervention or brake override. */
            entry;
            then driverDemand;

            state driverDemand {
                do action deliver : DeliverRequestedPropulsion;
            }

            state externalIntervention {
                do action reduce : ApplyExternalTorqueReduction;
            }

            state brakeOverride {
                do action overrideOnBrake : ApplyBrakeOverride;
            }

            transition enterInterventionByEcu
                first driverDemand accept req : TorqueInterventionRequest then externalIntervention;

            transition enterOverrideByBrakeFromDriver
                first driverDemand
                accept brake : BrakePedalPosition
                if brake.stopLightSwitchActive then brakeOverride;

            transition enterOverrideByBrakeFromIntervention
                first externalIntervention
                accept brake : BrakePedalPosition
                if brake.stopLightSwitchActive then brakeOverride;

            transition returnFromIntervention
                first externalIntervention accept rel : InterventionReleasedEvent then driverDemand;

            transition returnFromBrakeOverride
                first brakeOverride accept rel : InterventionReleasedEvent then driverDemand;
        }

        state safeState : SafeState {
            doc
            /* Confirmed fault: vehicle is held in a controllable safe state and the driver is warned optically. */
            entry action warn : WarnDriver;

            state torqueLimited : TorqueLimited {
                do action limitPower : LimitPropulsionPower;
            }
            state fuelCutOff : FuelCutOff {
                do action cutOffFuel : CutOffPropulsion;
            }
            state fullCutOff : FullCutOff {
                do action shutOffPower : CutOffPropulsion;
            }

            transition escalateToFuelCutOff
                first torqueLimited
                accept ev : SensorOrActuatorFaultEvent
                if ev.cutOffRequired then fuelCutOff;

            transition escalateToFullCutOff
                first fuelCutOff accept ev : ControllerFaultEvent then fullCutOff;
        }

        // Transitions from normalDriving (covers both driverDemand and torqueIntervention) to SafeState
        transition enterTorqueLimitedByExcess
            first normalDriving
            accept ev : UnintendedAccelerationEvent
            then safeState.torqueLimited;

        transition enterTorqueLimitedByExternalFault
            first normalDriving accept ev : ExternalRequestFaultEvent then safeState.torqueLimited;

        transition enterTorqueLimitedBySensorFault
            first normalDriving
            accept ev : SensorOrActuatorFaultEvent
            if not ev.cutOffRequired then safeState.torqueLimited;

        transition enterFuelCutOffByActuatorFault
            first normalDriving
            accept ev : SensorOrActuatorFaultEvent
            if ev.cutOffRequired then safeState.fuelCutOff;

        transition enterFullCutOffByControllerFault
            first normalDriving accept ev : ControllerFaultEvent then safeState.fullCutOff;
    }

    state l2Behavior : EgasBehavior;

    // Refinement Traceability: Behavior -> Runtime SHRs & All FSRs
    // (FunctionalSafetyComplianceRequirement is a development process req, not a runtime behavior)
    #refinement dependency behaviorRefinesRuntimeShrs
        from EgasBehavior
        to SteadyStatePrecisionRequirement, PedalResponseRequirement, TorqueInterventionRequirement;

    #refinement dependency behaviorRefinesAllFsrs
        from EgasBehavior
        to
            fsr01_PedalSensorPlausibility,
            fsr02_ThrottleSensorPlausibility,
            fsr03_SensorSystemFaultDetection,
            fsr04_ExternalTorqueRequestProtection,
            fsr05_ActuatorErrorDetection,
            fsr06_TorqueExcessSafeState,
            fsr07_FunctionControllerMonitoring,
            fsr08_BrakeOverride;

    // Action-level fine-grained refinement links
    #refinement dependency deliverRefinesPrecisionShr
        from DeliverRequestedPropulsion
        to SteadyStatePrecisionRequirement;
    #refinement dependency deliverRefinesResponseShr
        from DeliverRequestedPropulsion
        to PedalResponseRequirement;
    #refinement dependency reductionRefinesInterventionShr
        from ApplyExternalTorqueReduction
        to TorqueInterventionRequirement;
    #refinement dependency brakeOverrideRefinesFsr08 from ApplyBrakeOverride to fsr08_BrakeOverride;
    #refinement dependency rejectExternalRefinesFsr04
        from RejectFaultyExternalRequest
        to fsr04_ExternalTorqueRequestProtection;
    #refinement dependency limitPowerRefinesFsr03AndFsr06
        from LimitPropulsionPower
        to fsr03_SensorSystemFaultDetection, fsr06_TorqueExcessSafeState;
    #refinement dependency cutOffRefinesFsr05Fsr06Fsr07
        from CutOffPropulsion
        to
            fsr05_ActuatorErrorDetection,
            fsr06_TorqueExcessSafeState,
            fsr07_FunctionControllerMonitoring;
    #refinement dependency warnDriverRefinesFsr06 from WarnDriver to fsr06_TorqueExcessSafeState;
} // package EgasArch_Beh_Behavior

        bind engineSensor = estimateActualTorque.engineSensor;
        flow derivePermissibleTorque.permissibleTorque to compareTorque.permissibleTorque;
        flow estimateActualTorque.actualTorque to compareTorque.actualTorque;
        bind compareTorque.excessFlag = excessFlag;
    }

    action excessConfirmationWb : ExcessConfirmationFunction {
        action confirmExcessPersistence : ConfirmExcessPersistenceFunction;
        action issueSafeStateRequest : IssueSafeStateRequestFunction;
        bind excessFlag = confirmExcessPersistence.excessFlag;
        flow confirmExcessPersistence.confirmedExcess to issueSafeStateRequest.confirmedExcess;
        bind issueSafeStateRequest.safeStateRequest = safeStateRequest;
    }

    action safeStateEnforcementWb : SafeStateEnforcementFunction {
        action forceSafeStateDemand : ForceSafeStateDemandFunction;
        action raiseDriverWarning : RaiseDriverWarningFunction;
        bind safeStateRequest = forceSafeStateDemand.safeStateRequest;
        bind safeStateRequest = raiseDriverWarning.safeStateRequest;
        bind demand = forceSafeStateDemand.demand;
        bind forceSafeStateDemand.safeDemand = safeDemand;
        bind raiseDriverWarning.warning = warning;
    }

    action pedalSensorFaultDetectionWb : PedalSensorFaultDetectionFunction {
        action comparePedalChannels : ComparePedalChannelsFunction;
        action confirmPedalFault : ConfirmPedalFaultFunction;
        bind pedalPosition = comparePedalChannels.pedalPosition;
        flow comparePedalChannels.channelDeviated to confirmPedalFault.channelDeviated;
        bind confirmPedalFault.pedalFault = pedalFault;
    }

    action limpHomeTorqueLimitWb : LimpHomeTorqueLimitFunction {
        action evaluateLimpHomeCondition : EvaluateLimpHomeConditionFunction;
        action limitTorqueDemand : LimitTorqueDemandFunction;
        bind pedalFault = evaluateLimpHomeCondition.pedalFault;
        bind modeChannel = evaluateLimpHomeCondition.modeChannel;
        flow evaluateLimpHomeCondition.limpHomeActive to limitTorqueDemand.limpHomeActive;
        bind demand = limitTorqueDemand.demand;
        bind limitTorqueDemand.limitedDemand = limitedDemand;
    }

    action controllerMonitoringWb : ControllerMonitoringFunction {
        action checkQnaAnswer : CheckQnaAnswerFunction;
        action requestShutOff : RequestShutOffFunction;

        bind checkQnaAnswer.qnaQuestion = qnaQuestion;

    }

    action throttleSensorFaultDetectionWb : ThrottleSensorFaultDetectionFunction {

                     * definition level and are unaffected.
                     */
                }
                filter (
                    as SysML::ConnectionUsage hastype SysML::ConnectionUsage or
                    as SysML::AllocationUsage hastype SysML::AllocationUsage or
                    @ RequirementDerivation::derivation
                );
            }
        }

        abstract view def EditableMatrixView :> MatrixView {
            view :>> rowView {
                filter as SysML::Definition istype SysML::Definition;
            }
            view :>> columnView {
                filter as SysML::Definition istype SysML::Definition;
            }
            view :>> cellView {
                attribute :>> exposeFeaturesAndHeritage = true;
            }
        }

        view def ConnectionMatrixView :> EditableMatrixView {
            view :>> cellView {
                filter (
                    as SysML::ConnectionUsage hastype SysML::ConnectionUsage and
                    not (@ RequirementDerivation::derivation)
                );
            }
        }

        view def AllocationMatrixView :> EditableMatrixView {
            view :>> cellView {
                filter as SysML::AllocationUsage hastype SysML::AllocationUsage;
            }
        }

        view def RequirementMatrixView :> EditableMatrixView {
            view :>> cellView {
                filter @ RequirementDerivation::derivation;
            }
        }
    } // package <MVD> MatrixViewDefinitions

    package AbstractNavigationSteps {
        abstract calc def <Step> NavigationStep {
            doc
            /* The most generic navigation step: element in, collection out. */
            in elements [0..1];
            return targets [0..*]; // empty = navigation had no result
        }
        abstract calc def <Single> SinglevaluedStep :> Step {
            doc
            /* At most one target per input element, which may feed a further
             * step.
             */
        }
        abstract calc def <FanOut> FanOutStep :> Step {
            doc
            /* A step whose targets split the parent row into nested rows — the
             * 1:N join of the table model.
             *
             * A `*Fan` call must be the outermost call of a named view-level ref
             * (an *anchor*), never inline in a column or nested in another call:
             *
             *     ports = UsageFan(baseViewElement, UsageKind::portUsage);
             *
             * An anchor starts from `baseViewElement` or one other anchor, and
             * feeds at most one further anchor — anchors form a chain, not a
             * tree. A column then references one anchor and applies only non-Fan
             * steps to it. Declaration order does not matter, and columns sharing
             * an anchor are row-aligned.
             */
        }
        abstract calc def <List> ListElements :> Step {
            doc
            /* A terminal step: every target rendered together in one cell. Its
             * elements cannot feed another step, so a `*List` call may only be a
             * chain's LAST step. Such a column is read-only for every row.
             */
        }
        abstract calc def <First> FirstStep :> Single {
            doc
            /* For navigations that yield one target under the supported
             * modelling pattern, even though SysML permits more —
             * `DefinitionsFirst` for a usage typed by exactly one definition.
             * Not a way to take the head of a list: if more targets exist the
             * model deviates from the pattern, and the cell renders the first one
             * read-only, with a note.
             */
        }
    } // package AbstractNavigationSteps

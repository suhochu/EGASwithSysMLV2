package RV_Objectives {
    doc
    /* [목표(Goal) 뷰포인트]
     * EGAS 프로젝트가 존재하는 근본적인 이유를 정의한다: 어떤 문제(Problem Statement)를 풀기 위해
     * 어떤 시스템 아이디어(System Idea)를 제안하는지를 최상위 요구사항 계층으로 표현한다.
     * 이 패키지의 요구사항들은 이후 03/04번 파일의 이해관계자 우려사항(Concern) 및 이해관계자
     * 요구사항(Stakeholder Requirement)이 `:>` 특화(specialization)를 통해 근거로 삼는 최상위 앵커 역할을 한다.
     */

    private import ModelingMetadata::*;
    private import RV_StakeHolderModel::*;
    private import ReqCommon::*;

    @MetaInformation {
        version = "v0.1";
    }

    requirement def Objective {
        doc
        /* 프로젝트의 근본적인 문제 정의(Problem Statement), 시스템 아이디어(System Idea), 
         * 그리고 비즈니스 목표를 모델링하는 최상위 요구사항 타입
         */
    }

    // ==========================================
    // 1. Problem Statement (문제 정의)
    // ==========================================

    requirement def EgasProblemStatement :> Objective {
        doc
        /* 현대의 자동차는 운전자, 크루즈 컨트롤, 차체 자세 제어(ESP), 변속기 제어(TCU) 등 다양한 소스에서 동시다발적이고 상충되는 토크 요구가 발생한다.
         * 비정상적인 가속이나 센서 고장 등은 치명적인 사고로 직결되는 페인 포인트를 가진다.
         * 시스템은 다수의 복합적인 토크 요구를 안전하게 중재하고, 기계적 결함이나 가속/제동 동시 조작 상황에서도 차량의 추진력을 완벽하게 통제할 수 있는 고신뢰성 중앙 엔진 제어를 보장해야 한다.
         */

        @MetaInformation {
            version = "v0.1";
        }
    }

    // ================================================
    // 2. System Idea / Elevator Pitch (시스템 아이디어)
    // ================================================

    requirement def EgasSystemIdea :> EgasProblemStatement {
        doc
        /* EGAS(Electronic Throttle Control) 시스템은 차량 내 모든 토크 요구를 중앙집중식으로 수집하고 안전 최우선 원칙(예: Brake Override)에 따라 중재하는 지능형 엔진 제어 시스템이다.
         * 이중화된 센서 모니터링과 정밀한 공기, 연료, 점화 제어를 동기화하여 운전자의 의도를 정확하게 구현함과 동시에 극한의 주행 상황에서도 차량의 안정성을 보장해야 한다.
         */

        @MetaInformation {
            version = "v0.1";
        }
    }
} // package RV_Objectives


package RV_ContextModel {
    doc
    /* [시스템 컨텍스트 뷰포인트]
     * EGAS(관심 시스템)의 경계를 정의하는 어휘를 제공한다: 외부 액터, 주고받는 아이템(payload), 경계 포트,
     * 그리고 관심 시스템 자체(EgasSystem)를 블랙박스로, 시스템+외부 액터 전체를 컨텍스트(EgasContext)로 정의한다.
     * 내부 컴포넌트(EngineControlUnit/AccelerationPedal/ThrottleValve)는 05/06번 파일에서 그레이박스 유스케이스의
     * subject나 요구사항의 allocationBudget 대상으로 참조되지만, 이 파일의 EgasSystem 정의 자체는 포트만 노출하는
     * 순수 블랙박스이며 내부 배선(어떻게 연결되는지)은 "02. Functional Viewpoint"가 책임진다.
     */

    private import ReqCommon::*;
    public import RV_InterfaceModel::*;
    public import RV_OperationalModel::*;
    public import RV_ActorModel::*;
    private import ViewAndViewpintCommon::*;

    @MetaInformation {
        version = "v0.2";
    }

    // ==========================================
    // 1. 관심 시스템 (System of Interest) 정의
    // - 요구사항 뷰포인트에서는 블랙박스로만 취급: 포트(인터페이스)만 노출하고
    //   내부 구성(컴포넌트 분해 및 내부 흐름)은 기능/논리 뷰포인트에서 정의함
    // ==========================================

    part def EgasSystem {
        port ccPort : CruiseControlPort;
        port brakePort : BrakeInputPort;
        port torqueInterventionPort : TorqueInterventionPort;
        port sensorPort : SensorInputPort;
        port fuelCmdPort : FuelCommandPort;
        port combustionStatusPort : CombustionStatusPort;
        port accelPort : AccelerationInputPort;
        port airIntakePort : AirIntakePort;
        port calibrationPort : CalibrationPort;
        port diagPort : DiagnosticPort;
        port ignitionPort : IgnitionPort;
        port dashboardPort : DashboardPort;

        // 복합(Composition) 관계: 내부 구성요소 존재만 이 레벨에서 선언하고, 배선/연결은 기능 뷰포인트("02. Functional Viewpoint")가 책임진다
        part ecu : EngineControlUnit;
        part pedal : AccelerationPedal;
        part throttle : ThrottleValve;

        event occurrence commandsManualTorque; // 수동 목표 토크 산출 및 제어 명령
        event occurrence commandsCruiseTorque; // 크루즈 목표 토크 자동 산출 및 제어 명령
        event occurrence overridesAndCutsTorque; // 브레이크 입력에 따른 기존 요구 무효화 및 동력 차단

        event occurrence commandsEmergencyReduction; // 정상 긴급 감속 명령
        event occurrence compensatesCommDelay; // 지연 보상 제어 명령
        event occurrence commandsEmergencyFuelCut; // 예외 상황 연료 차단 명령

        event occurrence compensatesDelayAndFriction; // 정상 지연/마찰 보정 명령
        event occurrence triggersEmergencyFailSafe; // 결함 감지 및 페일세이프 차단 명령

        event occurrence authorizesAccess; // 유효 메모리 접근 인가
        event occurrence verifiesAndCommitsData; // 데이터 무결성 검증 및 기록
        event occurrence rejectsAndRollsBack; // 비유효 접근 거절 및 롤백

        event occurrence registersSensorFault; // 센서 결함 상태 기록
        event occurrence processesDtcRequest; // DTC 조회 요청 처리 및 응답
        event occurrence processesZeroPointCal; // 영점 조정 처리 및 응답
        event occurrence rejectsInvalidRequest; // 비유효 진단 요청 거절

        // 시스템이 해당 운용 요구사항 기능들을 수행함을 명시
        perform action manageTorque
            : EgasOperationalRequirements_UC1::ManageDrivingTorqueArbitration;
        perform action manageEmergency
            : EgasOperationalRequirements_UC2::ManageEmergencyIntervention;
        perform action manageFailSafe : EgasOperationalRequirements_UC3::ManageHardwareFailSafe;
        perform action manageCalibration
            : EgasOperationalRequirements_UC4::ManagePowertrainCalibration;
        perform action manageDiagnostics
            : EgasOperationalRequirements_UC5::ManageDiagnosticsAndMaintenance;

        action def EgasMainContinuousOperation {
            doc
            /* 시스템 전원이 켜진 상태에서 모든 UC 운용 로직을 병렬로 무한 반복 수행 */

            then action uc1 : EgasOperationalRequirements_UC1::ManageDrivingTorqueArbitration;
            then action uc2 : EgasOperationalRequirements_UC2::ManageEmergencyIntervention;
            then action uc3 : EgasOperationalRequirements_UC3::ManageHardwareFailSafe;
            then action uc4 : EgasOperationalRequirements_UC4::ManagePowertrainCalibration;
            then action uc5 : EgasOperationalRequirements_UC5::ManageDiagnosticsAndMaintenance;

            fork mainLoop;
                then uc1;
                then uc2;
                then uc3;
                then uc4;
                then uc5;
        }

        exhibit state egasBehavior : RV_BehavioralModel::EgasBehavior;

        @MetaInformation {
            version = "v0.5";
        }
    }

    // ==========================================
    // 2. 시스템 컨텍스트 (System Context)
    // ==========================================

    part def EgasContext {
        // [Step 1] 외부 환경(액터) 및 관심 시스템 인스턴스화
        part driver : Driver;
        part ccLever : CruiseControlLever [0..1];
        part brake : Brake;
        part addEcu : AddEcu;
        part sensors : AddSensorComponents;
        part fuelSys : FuelInjectionIgnitionSystem;
        part engine : Engine;
        part calTool : CalibrationTool;
        part diagTool : DiagnosticTool;
        part powerSys : VehiclePowerSystem;
        part dashboard : DashboardHMI;

        part egas : EgasSystem;

        // [Step 2] 명시적인 구조적 연결 (Connections)
        // 외부 액터와 EgasSystem의 포트 간의 상호작용 경로(경계 인터페이스)를 공식적으로 개통합니다.
        connect driver.driverPort to egas.accelPort {
            flow of AccelerationIntent from driver.driverPort to egas.accelPort;
        }
        connect ccLever.ccPort to egas.ccPort {
            flow of CruiseControlCommand from ccLever.ccPort to egas.ccPort;
        }
        connect brake.brakePort to egas.brakePort {
            flow of BrakePedalPosition from brake.brakePort to egas.brakePort;
        }
        connect addEcu.addEcuPort to egas.torqueInterventionPort {
            flow of TorqueInterventionRequest from addEcu.addEcuPort to egas.torqueInterventionPort;
        }
        connect sensors.sensorPort to egas.sensorPort {
            flow of EngineSensorData from sensors.sensorPort to egas.sensorPort;
        }

        // fuelSys는 명령어 수신용 포트와 상태 송신용 포트 두 곳에 모두 연결됩니다.
        connect fuelSys.cmdPort to egas.fuelCmdPort {
            flow of FuelIgnitionCommand from egas.fuelCmdPort to fuelSys.cmdPort;
        }
        connect fuelSys.statusPort to egas.combustionStatusPort {
            flow of CombustionSystemStatus from fuelSys.statusPort to egas.combustionStatusPort;
        }
        connect engine.intakePort to egas.airIntakePort {
            flow of IntakeAirFlow from egas.airIntakePort to engine.intakePort;
        }

        // 툴과의 양방향 통신 경로 연결
        connect calTool.calPort to egas.calibrationPort {
            flow of CalibrationRequest from calTool.calPort to egas.calibrationPort;
            flow of CalibrationStatus from egas.calibrationPort to calTool.calPort;
        }
        connect diagTool.diagPort to egas.diagPort {
            flow of DiagnosticRequest from diagTool.diagPort to egas.diagPort;
            flow of DiagnosticStatus from egas.diagPort to diagTool.diagPort;
        }

        connect powerSys.ignitionPort to egas.ignitionPort {
            flow of IgnitionState from powerSys.ignitionPort to egas.ignitionPort;
        }
        connect egas.dashboardPort to dashboard.dashboardPort {
            flow of DashboardWarning from egas.dashboardPort to dashboard.dashboardPort;
        }

        @MetaInformation {
            version = "v0.7";
        }
    } // package EgasSystemContext
} // package RV_ContextModel


package RV_ActorModel {
    private import ReqCommon::*;
    private import RV_InterfaceModel::*;

    @MetaInformation {
        version = "v0.2";
    }

    // ==========================================
    // 1. 내부 컴포넌트 정의 (Internal Components)
    // - 관심 시스템 내부의 요소로, 그레이박스 유스케이스(05번 파일)와 요구사항 할당(06번 파일)의 대상이 됨
    // ==========================================

    part def EngineControlUnit {
        // 시스템의 중앙 마이크로컨트롤러(µC)로서 가속 페달, 크루즈 컨트롤,
        // 외부 제어기의 토크 요구(Torque Request)를 수집 및 중재(Arbitration)함.
        // 목표 토크를 달성하기 위해 스로틀 밸브 및 연료/점화 시스템에 제어 명령을 내리며,
        //  오작동을 감지하는 안전 모니터링을 수행함.

        @MetaInformation {
            version = "v0.2";
        }
    }

    part def AccelerationPedal {
        // 운전자의 물리적인 가속 의지(페달을 밟는 정도)를 전기적 신호로 변환하여 ECU에 1차적인 운전자 토크 요구량으로 전달함.

        @MetaInformation {
            version = "v0.2";
        }
    }

    part def ThrottleValve {
        // ECU의 제어 명령에 따라 개방 각도를 조절하여 엔진으로 유입되는 공기량을 물리적으로 제어함.
        // 동시에 현재 밸브의 위치 값을 ECU로 피드백하여 정밀한 폐루프 제어(Closed-loop control)를 가능하게 함.

        @MetaInformation {
            version = "v0.2";
        }
    }

    // ==========================================
    // 2. 외부 액터 정의 (External Actors)
    // ==========================================

    part def Driver {
        // 가속 페달을 밟아 가속 의도를 표현하는 물리적 운전자. AccelerationPedal(센서)이 EgasSystem
        // 내부 컴포넌트로 포함되므로, 운전자의 물리적 가속 의도가 시스템 경계를 넘는 입력이 된다.
        // (03번 파일의 stakeholder Driver와는 별개 개념: 이해관계자 관점이 아닌 컨텍스트 경계 관점의 액터)

        port driverPort : ~AccelerationInputPort;
        event occurrence pressesAccelPedal; // 가속 페달 입력 (가속 의지)

        @MetaInformation {
            version = "v0.3";
        }
    }

    part def CruiseControlLever {
        // 크루즈 컨트롤(Cruise Control) 활성화를 위한 운전자의 조작 상태를 전달하며,
        // 지정된 속도를 유지하기 위한 지속적인 토크 요구 명령을 시스템에 입력함.

        port ccPort : ~CruiseControlPort;
        event occurrence setsCruiseSpeed; // 크루즈 컨트롤 목표 속도 설정

        @MetaInformation {
            version = "v0.3";
        }
    }

    part def Brake {
        // 운전자의 제동 의지를 ECU에 전달함.
        // 가속 페달 신호와 브레이크 신호가 동시에 입력될 경우,
        // 제동을 우선시하여 토크를 강제로 차단하거나 줄이는 안전 기능(Brake Override)을 트리거함.

        port brakePort : ~BrakeInputPort;
        event occurrence pressesBrakePedal; // 브레이크 페달 입력 (제동 의지)

        @MetaInformation {
            version = "v0.3";
        }
    }

    part def AddEcu {
        // 변속기 제어기(TCU), 차체 자세 제어기(ESP) 등 다른 시스템의 마이크로컨트롤러가
        // 전송하는 토크 개입(Torque Intervention) 신호를 전달함.
        // 기어 변속 시점의 토크 저감이나 미끄럼 방지를 위한 토크 제한 등을 요구함.

        port addEcuPort : ~TorqueInterventionPort;
        event occurrence requestsIntervention; // 토크 개입 요청 (정상/지연/고착/해제 통합)

        @MetaInformation {
            version = "v0.3";
        }
    }

    part def AddSensorComponents {
        // 차량의 종방향 가속도 변화율(Jerk),엔진 회전수(RPM), 냉각수 온도, 흡기 온도 등 ECU가 목표 토크를
        // 정확하게 계산하고 보정하는 데 필요한 주변 환경 및 물리적 상태 데이터를 제공함.

        port sensorPort : ~SensorInputPort;
        event occurrence sendsNormalSensorData; // 정상 범위의 센서 데이터 송신
        event occurrence sendsFaultyData; // 물리적 결함(온도 초과 등) 데이터 송신
        event occurrence sendsDynamicsFeedback; // 차량 거동(Jerk 등) 피드백 송신

        @MetaInformation {
            version = "v0.3";
        }
    }

    part def FuelInjectionIgnitionSystem {
        // 스로틀 밸브의 공기량 제어와 협조하여,
        // ECU가 요구하는 최종 토크를 실제 엔진 동력으로 구현하기 위해 연료 분사량과 점화 시기를 제어하는 외부 구동계 역할을 수행함.

        port cmdPort : ~FuelCommandPort;
        port statusPort : ~CombustionStatusPort;

        event occurrence injectsFuel; // 연료/점화 제어 실행
        event occurrence executesReducedFuel; // 축소된 연료/점화 지각 실행
        event occurrence executesMaintainedFuel; // 동결(유지)된 연료량 실행
        event occurrence executesDriverBasedFuel; // 개입 해제 후 운전자 요구 기반 연료/점화 실행

        event occurrence executesFuelCut; // 연료 차단 실행 (제동 오버라이드 및 페일세이프 통합)
        event occurrence executesCompensatedFuel; // 보정된 연료/점화 제어 실행
        event occurrence sendsCombustionFeedback; // 연소 상태(실화율) 피드백 송신

        @MetaInformation {
            version = "v0.3";
        }
    }

    part def Engine {
        // 스로틀 밸브(ThrottleValve)를 통해 유입된 공기를 받아들이는 물리적 엔진 블록(실린더/흡기 매니폴드).
        // FuelInjectionIgnitionSystem과는 별도의 외부 액터로, 공기 경로(air path)만을 담당함.

        port intakePort : ~AirIntakePort;
        event occurrence receivesAirFlow; // 공기 유입 수용
        event occurrence receivesAirCut; // 공기 차단 수용
        event occurrence receivesReducedAir; // 축소된 공기량 수신
        event occurrence receivesMaintainedAir; // 동결(유지)된 공기량 수신
        event occurrence receivesDriverBasedAir; // 개입 해제 후 운전자 요구 기반 공기량 수신
        event occurrence receivesCompensatedAir; // 보정된 공기량 수신

        @MetaInformation {
            version = "v0.3";
        }
    }

    part def CalibrationTool {
        // 엔지니어가 사용하는 외부 XCP/CCP 튜닝 장비
        port calPort : ~CalibrationPort;
        event occurrence requestsValidAccess; // 유효한 메모리 주소 접근 요청
        event occurrence receivesAuthOk; // 인가 응답(200) 수신
        event occurrence sendsTuningData; // 튜닝 파라미터 데이터 송신
        event occurrence receivesVerifiedChecksum; // 검증된 체크섬 수신
        event occurrence requestsInvalidAccess; // 비유효/잘못된 메모리 주소 접근 요청
        event occurrence receivesErrorResponse; // 거절/에러 응답 수신

        @MetaInformation {
            version = "v0.3";
        }
    }

    part def DiagnosticTool {
        // 서비스 센터(A/S) 엔지니어가 사용하는 외부 진단 장비

        attribute effort : ElicitationEffortKind; // 요구사항을 도출하고 시스템 설계에 반영하기 위해 필요한 노력(시간, 자원, 복잡도) 기준

    enum def StakeholderPriorityKind {
        Critical; // 요구사항 누락 시 프로젝트가 즉각 실패하거나 법적/안전상 치명적 제재를 받는 핵심 주체 (예: 규제 기관, 투자자, 핵심 의사결정권자).
        High; // 시스템 수용성과 비즈니스 가치에 직접적인 영향을 미치는 주요 주체 (예: 주 사용층, 도메인 전문가).
        Medium; // 시스템과 상호작용하지만 프로젝트 방향성을 일방적으로 변경할 권한은 없는 주체 (예: 2차 사용자, 간접 연계 시스템 담당자).
        Low; // 시스템 결과에 관심은 있으나 요구사항 반영 우선순위가 가장 낮은 주체.
    }

    enum def ElicitationEffortKind {
        High; // 요구사항이 매우 모호하거나, 상충되는 의견을 조율하기 위해 다수의 워크샵과 프로토타이핑이 필요한 경우.
        Medium; // 표준적인 인터뷰, 설문, 기존 문서 분석을 통해 요구사항을 도출하고 설계에 반영할 수 있는 일반적인 수준.
        Low; // 이미 명확한 국제 표준(예: ISO 26262), 법규, 또는 기존 레거시 시스템의 사양서가 존재하여 요구사항을 단순히 추출 및 차용하면 되는 경우.
    }

    enum def StakeholderCategoryKind {
        EndUser; // 시스템을 직접 조작하고 경험하는 주체 (예: 운전자, 승객).

        AcquirerCustomer; // 시스템 개발을 의뢰하거나 비즈니스적 가치를 평가하는 주체 (예: 완성차 OEM, 제품 관리자(PM)).
        SupplierPartner; // 시스템에 연동되는 외부 부품이나 하위 모듈을 제공하는 주체 (예: Tier 1/2 센서 공급사, 변속기(TCU) 제조사).
        MaintenanceSupport; // 시스템 출시 후 수리, 정비, 업데이트를 담당하는 주체 (예: 정비소 미캐닉, 서비스 센터).

package RV_UseCases {
    private import RV_ContextModel::*;
    private import RV_StakeholderRequirementModel::*;

    private import SI::*;

    constraint validationCriteria;

    use case def <UC1> 'Driving Torque Demand Arbitration & Control' {
        @MetaInformation {
            version = "v0.2";
        }

        doc
        /* Arbitrates driver/CC torque demands (incl. Brake Override), 
         * outputs engine/fuel commands, and receives feedback.
         */

        subject egas : EgasSystem;

        // Input actors
        actor driver : Driver;
        actor ccLever : CruiseControlLever;
        actor brake : Brake;

        // Output/Feedback actors
        actor fuelSys : FuelInjectionIgnitionSystem;
        actor engine : Engine;
        actor dynamicsSensor : AddSensorComponents;

        satisfy requirement 'SHR-01'; // reqSafeTorqueArbitration
        satisfy requirement 'SHR-08'; // reqDriverPedalResponsiveness
        satisfy requirement 'SHR-02-1'; // reqSteadyStatePrecision

        // === Action Scenarios ===

        action UC1_Normal_Manual_Acceleration {
            doc
            /* [Main] Processes pedal input < 50ms, calculates torque, 
             * and commands engine/fuel for smooth acceleration (Jerk/Overshoot controlled).
             */

            constraint precondition {
                doc
                /* Driver physically pressed accel pedal. */
                egas.accelPort.accelIntent.pedalDisplacement > 0.0 [mm]
            }

            // Messages
            message msgAccel of AccelerationIntent
                from driver.pressesAccelPedal to egas.commandsManualTorque {
                bind payload = egas.accelPort.accelIntent;
            }

            message msgAirCmd of IntakeAirFlow
                from egas.commandsManualTorque to engine.receivesAirFlow {
                bind payload = egas.airIntakePort.airFlow;
            }

            message msgFuelCmd of FuelIgnitionCommand
                from egas.commandsManualTorque to fuelSys.injectsFuel {
                bind payload = egas.fuelCmdPort.fuelCmd;
            }

            // Feedback & non-functional constraints
            message msgFeedback of CombustionSystemStatus
                from fuelSys.injectsFuel to egas.commandsManualTorque {
                bind payload = egas.combustionStatusPort.combustionStatus;
                ref constraint <uc1_vc02>;
            }

            message msgDynamics of VehicleDynamicsFeedback
                from dynamicsSensor.sendsDynamicsFeedback to egas.commandsManualTorque {
                bind payload = egas.sensorPort.dynamicsStatus;
                ref constraint <uc1_vc03>;
            }

            // Sequence Flow
            first msgAccel;

            // Output air/fuel commands in parallel
            then fork commandOut;
                then msgAirCmd;
                then msgFuelCmd;

                    first msgAirCmd then j;
                    first msgFuelCmd then j;

            // Receive combustion and dynamics feedback in parallel
            join j;
            then fork feedbackIn;
                then msgFeedback;
                then msgDynamics;

            // Verification Criterias
            constraint <uc1_vc01> responseTimeConstraint :> validationCriteria {
                doc
                /* Response: Air/fuel command starts within 50ms of accel input. */
                msgAirCmd.start - msgAccel.done <= 50.0 [milli * s] and
                msgFuelCmd.start - msgAccel.done <= 50.0 [milli * s]
            }

            constraint <uc1_vc02> torqueOvershootConstraint :> validationCriteria {
                doc
                /* Precision: Torque overshoot <= 5.0%. */
                @PlantObservation;
                msgFeedback.payload.torqueOvershoot <= 5.0
            }

            constraint <uc1_vc03> rideComfortJerkConstraint :> validationCriteria {
                doc
                /* Ride Comfort: Longitudinal jerk <= 10.0. */
                @PlantObservation;
                msgDynamics.payload.longitudinalJerk <= 10.0
            }

            constraint postcondition {
                doc
                /* Commands applied; overshoot and jerk within limits. */
                @PlantObservation;
                msgFeedback.payload.torqueOvershoot <= 5.0 and
                msgDynamics.payload.longitudinalJerk <= 10.0
            }
        }

        action UC1_Alternative_Cruise_Control_Driving {
            /* [Alt] CC calculates speed error and automatically commands 
             * torque via closed-loop control to maintain set speed.
             */
            constraint precondition {
                doc
                /* CC target speed is set and active. */
                egas.ccPort.ccCmd.setSpeed > 0.0 [km / h]
            }

            // Messages
            message msgCcReq of CruiseControlCommand
                from ccLever.setsCruiseSpeed to egas.commandsCruiseTorque {
                bind payload = egas.ccPort.ccCmd;
            }

            message msgAirCmd_CC of IntakeAirFlow
                from egas.commandsCruiseTorque to engine.receivesAirFlow {
                bind payload = egas.airIntakePort.airFlow;
            }

            message msgFuelCmd_CC of FuelIgnitionCommand
                from egas.commandsCruiseTorque to fuelSys.injectsFuel {
                bind payload = egas.fuelCmdPort.fuelCmd;
            }

            message msgFeedback_CC of CombustionSystemStatus
                from fuelSys.injectsFuel to egas.commandsCruiseTorque {
                bind payload = egas.combustionStatusPort.combustionStatus;
                ref constraint <uc1_vc04>;
            }

            // Sequence Flow
            first msgCcReq;

            fork commandOut;
                then msgAirCmd_CC;
                then msgFuelCmd_CC;

                    first msgAirCmd_CC then j;
                    first msgFuelCmd_CC then j;

            join j;
            then msgFeedback_CC;

            // [Verification Criterias
            constraint <uc1_vc04> steadyStateErrorConstraint :> validationCriteria {
                doc
                /* Precision: Steady-state speed error <= 2.0%. */
                msgFeedback_CC.payload.steadyStateError <= 2.0
            }

            constraint postcondition {
                doc
                /* Target speed maintained with steady-state error <= 2.0%. */
                msgFeedback_CC.payload.steadyStateError <= 2.0
            }
        }

        action UC1_Exception_Brake_Override_Intervention {
            doc
            /* [Exception] Brake input overrides all acceleration demands; 
             * air/fuel cut off immediately for safety.
             */
            constraint precondition {
                doc
                /* Brake pedal input detected. */
                egas.brakePort.brakeInput.pressure > 0.0 [Pa]
            }

            // Messages
            message msgBrakeInput of BrakePedalPosition
                from brake.pressesBrakePedal to egas.overridesAndCutsTorque {
                bind payload = egas.brakePort.brakeInput;
            }

            message msgAirCutCmd of IntakeAirFlow
                from egas.overridesAndCutsTorque to engine.receivesAirCut {
                bind payload = egas.airIntakePort.airFlow;
                ref constraint <uc1_vc06>;
            }

            message msgFuelCutCmd of FuelIgnitionCommand
                from egas.overridesAndCutsTorque to fuelSys.executesFuelCut {
                bind payload = egas.fuelCmdPort.fuelCmd;
                ref constraint <uc1_vc07>;
            }

            // Sequence Flow
            first msgBrakeInput;
            fork commandOut;
                then msgAirCutCmd;
                then msgFuelCutCmd;

            // Verification Criterias
            constraint <uc1_vc05> overrideResponseTimeConstraint :> validationCriteria {
                doc
                /* Response: Override and issue cut-off commands within 10ms of brake input. */
                msgAirCutCmd.start - msgBrakeInput.done <= 10.0 [milli * s] and
                msgFuelCutCmd.start - msgBrakeInput.done <= 10.0 [milli * s]
            }

            constraint <uc1_vc06> airCutOffFailSafe :> validationCriteria {
                doc
                /* Failsafe: Air intake cut off. */
                msgAirCutCmd.payload.massFlow <= 0.0 [kg / s]
            }

            constraint <uc1_vc07> fuelCutOffFailSafe :> validationCriteria {
                doc
                /* Failsafe: Fuel injection cut off. */
                msgFuelCutCmd.payload.injectionPulseWidth <= 0.0 [s]
            }

            constraint postcondition {
                doc
                /* Air and fuel cut off; system safe. */
                msgAirCutCmd.payload.massFlow <= 0.0 [kg / s] and
                msgFuelCutCmd.payload.injectionPulseWidth <= 0.0 [s]
            }
        }

        // Interaction Flows

        // 1. Normal Scenario Flows
        flow from driver to UC1_Normal_Manual_Acceleration;
        flow from UC1_Normal_Manual_Acceleration to engine;
        flow from UC1_Normal_Manual_Acceleration to fuelSys;
        flow from fuelSys to UC1_Normal_Manual_Acceleration;
        flow from dynamicsSensor to UC1_Normal_Manual_Acceleration;

        // 2. Alternative Scenario Flows
        flow from ccLever to UC1_Alternative_Cruise_Control_Driving;
        flow from UC1_Alternative_Cruise_Control_Driving to engine;
        flow from UC1_Alternative_Cruise_Control_Driving to fuelSys;
        flow from fuelSys to UC1_Alternative_Cruise_Control_Driving;

        // 3. Exception Scenario Flows
        flow from brake to UC1_Exception_Brake_Override_Intervention;
        flow from UC1_Exception_Brake_Override_Intervention to engine;
        flow from UC1_Exception_Brake_Override_Intervention to fuelSys;
    }

    use case def <UC2> 'Emergency Intervention' {
        doc
        /* 빙판길 미끄럼 등 긴급 상황에서 ESP/TCU의 토크 저감 요청 시, 
         * 스로틀 공기량 제어와 연료/점화 지각 제어를 병렬로 수행하여 
         * 15ms 이내에 즉각적으로 물리적 토크를 저감하는 유스케이스
         */

        subject egas : EgasSystem;

        // 입력 액터
        actor addEcu : AddEcu; // ESP, TCU 등 외부 제어기

        // 출력 액터
        actor engine : Engine;
        actor fuelSys : FuelInjectionIgnitionSystem;

        satisfy requirement 'SHR-04-1'; // reqTorqueInterventionByAirflow
        satisfy requirement 'SHR-04-2'; // reqTorqueInterventionByFuelIgnition
        satisfy requirement 'SHR-04-3'; // reqTorqueInterventionFallbackFuelOnly (OR 대체 경로, UC2_Exception에서 실현)

        // ==========================================
        // 액션(시나리오) 정의 - 각 액션이 하나의 완전한 시나리오를 대변함
        // ==========================================

        action UC2_Normal_Emergency_Torque_Reduction {
            doc
            /* [주 성공 시나리오] 
             * ESP/TCU의 긴급 토크 저감 요청을 수신하여 15ms 이내에 목표 공기량을 축소함과 동시에, 
             * 연료 인젝션 펄스 단축 및 점화 시기를 지각시켜 신속하게 토크를 저감하는 전체 정상 제어 흐름
             */
            constraint precondition {
                doc
                /* ESP/TCU 등 외부 제어기에서 미끄럼 방지 등을 위한 토크 저감 요구(음수 값)를 전달한 상태여야 함 
                 * 수신 지연 15ms 이하의 정상 통신 상태여야 함
                 */
                egas.torqueInterventionPort.torqueReq.requestedTorqueDelta < 0.0 [N * m] and
                egas.torqueInterventionPort.torqueReq.latency <= 15.0 [milli * s]
            }

            // [Messages]
            message msgInterventionReq of TorqueInterventionRequest
                from addEcu.requestsIntervention to egas.commandsEmergencyReduction {
                bind payload = egas.torqueInterventionPort.torqueReq;
            }

            message msgEmergAirCmd of IntakeAirFlow
                from egas.commandsEmergencyReduction to engine.receivesReducedAir {
                bind payload = egas.airIntakePort.airFlow;
                ref constraint <uc2_vc02>;
            }

            message msgEmergFuelCmd of FuelIgnitionCommand
                from egas.commandsEmergencyReduction to fuelSys.executesReducedFuel {
                bind payload = egas.fuelCmdPort.fuelCmd;
                ref constraint <uc2_vc03>;
            }

            // [Sequence Flow]
            first msgInterventionReq;

            // 토크 저감 요청 수신 후, 공기량 축소와 연료 분사 단축을 15ms 이내 병렬 하달
            fork commandOut;
                then msgEmergAirCmd;
                then msgEmergFuelCmd;

            // [Verification Criterias]
            constraint <uc2_vc01> emergencyResponseTime :> validationCriteria {
                doc
                /* [VC#01] 긴급 개입 응답 시간: 
                 * 토크 저감 요청 수신 후 15ms 이내에 공기량 및 연료 제어 명령이 병렬 하달되어야 함
                 */
                msgEmergAirCmd.start - msgInterventionReq.done <= 15.0 [milli * s] and
                msgEmergFuelCmd.start - msgInterventionReq.done <= 15.0 [milli * s]
            }

            constraint <uc2_vc02> airReductionConstraint :> validationCriteria {
                doc
                /* 공기량 즉각 축소: 직전 주기보다 물리적 공기량이 작아야 함 */
                msgEmergAirCmd.payload.massFlow < msgEmergAirCmd.payload.previousMassFlow
            }

            constraint <uc2_vc03> fuelReductionConstraint :> validationCriteria {
                doc
                /* 연료량 즉각 축소: 직전 주기보다 인젝션 펄스가 짧아야 함 */
                msgEmergFuelCmd.payload.injectionPulseWidth <
                msgEmergFuelCmd.payload.previousPulseWidth
            }

            constraint postcondition {
                doc
                /* 긴급 개입 종료 후 공기량과 연료량이 모두 직전 주기 대비 감소되어 토크가 저감된 상태여야 함 */
                msgEmergAirCmd.payload.massFlow < msgEmergAirCmd.payload.previousMassFlow and
                msgEmergFuelCmd.payload.injectionPulseWidth <
                msgEmergFuelCmd.payload.previousPulseWidth
            }
        }

        action UC2_Alternative_Communication_Delay_Compensation {
            doc
            /* [대안 시나리오] 
             * 외부 제어기와의 통신 지연으로 일시적인 토크 저감 신호 누락 발생 시, 
             * 타임아웃 전까지 직전의 유효한 토크 저감 요구값을 기반으로 자체 보상 제어를 유지하는 흐름
             */

            constraint precondition {
                doc
                /* 외부 제어기와의 통신 지연 또는 노이즈로 인해 토크 저감 요청 신호의 수신 지연이 15ms를 초과한 상태여야 함 */
                egas.torqueInterventionPort.torqueReq.latency > 15.0 [milli * s]
            }

            // [Messages]
            message msgInterventionReq_Delay of TorqueInterventionRequest
                from addEcu.requestsIntervention to egas.compensatesCommDelay {
                bind payload = egas.torqueInterventionPort.torqueReq;
            }

            message msgMaintainAirCmd of IntakeAirFlow
                from egas.compensatesCommDelay to engine.receivesMaintainedAir {
                bind payload = egas.airIntakePort.airFlow;
                ref constraint <uc2_vc04>;
            }

            message msgMaintainFuelCmd of FuelIgnitionCommand
                from egas.compensatesCommDelay to fuelSys.executesMaintainedFuel {
                bind payload = egas.fuelCmdPort.fuelCmd;
                ref constraint <uc2_vc05>;
            }

            // [Sequence Flow]
            first msgInterventionReq_Delay;
            fork commandOut;
                then msgMaintainAirCmd;
                then msgMaintainFuelCmd;

            // [Verification Criterias]
            constraint <uc2_vc04> airMaintenanceConstraint :> validationCriteria {
                doc
                /* 공기량 동결: 통신 지연 시 직전 주기의 공기량을 그대로 유지 */
                msgMaintainAirCmd.payload.massFlow == msgMaintainAirCmd.payload.previousMassFlow
            }

            constraint <uc2_vc05> fuelMaintenanceConstraint :> validationCriteria {
                doc
                /* 연료량 동결: 통신 지연 시 직전 주기의 연료량을 그대로 유지 */
                msgMaintainFuelCmd.payload.injectionPulseWidth ==
                msgMaintainFuelCmd.payload.previousPulseWidth
            }

            constraint postcondition {
                doc
                /* 통신 지연 보상 종료 후 공기량과 연료량이 직전 주기 값과 동일하게 유지되어 급격한 토크 변화가 없는 상태여야 함 */
                msgMaintainAirCmd.payload.massFlow == msgMaintainAirCmd.payload.previousMassFlow and
                msgMaintainFuelCmd.payload.injectionPulseWidth ==
                msgMaintainFuelCmd.payload.previousPulseWidth
            }
        }

        action UC2_Alternative_Intervention_Release {
            doc
            /* [대안 시나리오]
             * 외부 제어기가 토크 저감 해제(requestedTorqueDelta >= 0)를 요청하면 긴급 개입 상태를 종료하고,
             * 공기/연료 명령을 운전자 요구 기반 제어로 복귀시키는 흐름 (스로틀 고착 래치가 없는 경우)
             */

            attribute releaseResponseLimit :> ISQ::duration; // [TBD] 섀시 제어 부서와 합의 후 값 입력

            constraint precondition {
                doc
                /* 긴급 개입이 활성 상태이며, 외부 제어기가 해제(토크 저감량 0 이상)를 요청한 상태여야 함 */
                egas.torqueInterventionPort.torqueReq.requestedTorqueDelta >= 0.0 [N * m]
            }

            // [Messages]
            message msgReleaseReq of TorqueInterventionRequest
                from addEcu.requestsIntervention to egas.releasesIntervention {
                bind payload = egas.torqueInterventionPort.torqueReq;
            }

            message msgResumedAirCmd of IntakeAirFlow
                from egas.releasesIntervention to engine.receivesDriverBasedAir {
                bind payload = egas.airIntakePort.airFlow;
                ref constraint <uc2_vc08>;
            }

            message msgResumedFuelCmd of FuelIgnitionCommand
                from egas.releasesIntervention to fuelSys.executesDriverBasedFuel {
                bind payload = egas.fuelCmdPort.fuelCmd;
                ref constraint <uc2_vc09>;
            }

            // [Sequence Flow]
            first msgReleaseReq;
            fork commandOut;
                then msgResumedAirCmd;
                then msgResumedFuelCmd;

            // [Verification Criterias] (EGAS 응답시간: 수신 → 출력 시각 차이)
            constraint <uc2_vc08> airResumeTimeConstraint :> validationCriteria {
                doc
                /* 해제 요청 수신 후 공기량 명령이 운전자 요구 기반으로 복귀하기까지의 EGAS 응답시간 */
                msgResumedAirCmd.start - msgReleaseReq.done <= releaseResponseLimit
            }

            constraint <uc2_vc09> fuelResumeTimeConstraint :> validationCriteria {
                doc
                /* 해제 요청 수신 후 연료/점화 명령이 운전자 요구 기반으로 복귀하기까지의 EGAS 응답시간 */
                msgResumedFuelCmd.start - msgReleaseReq.done <= releaseResponseLimit
            }

            constraint postcondition {
                doc
                /* 해제 후 긴급 개입이 종료되고 연료 차단 없이 운전자 요구 기반 제어로 복귀한 상태여야 함 */
                msgResumedAirCmd.start - msgReleaseReq.done <= releaseResponseLimit and
                msgResumedFuelCmd.start - msgReleaseReq.done <= releaseResponseLimit
            }
        }

        action UC2_Exception_Actuator_Stuck_Fuel_Cut {
            doc
            /* [예외 시나리오] 
             * 긴급 개입 중 스로틀 밸브 고착(Stuck) 등 하드웨어 오류로 공기량 축소가 불가능함이 감지될 경우, 
             * 즉시 공기량 제어 시도를 포기하고 엔진 보호를 위해 연료 시스템을 전면 차단(Fuel Cut)하는 흐름
             */
            constraint precondition {
                doc
                /* 긴급 개입 시도에도 불구하고 스로틀 공기량이 줄어들지 않아 밸브 고착(Stuck)이 강력히 의심되는 상태여야 함 */
                egas.airIntakePort.airFlow.massFlow >= egas.airIntakePort.airFlow.previousMassFlow
            }

            // [Messages]
            message msgInterventionReq_Stuck of TorqueInterventionRequest
                from addEcu.requestsIntervention to egas.commandsEmergencyFuelCut {
                bind payload = egas.torqueInterventionPort.torqueReq;
            }

            message msgFuelCutCmd of FuelIgnitionCommand
                from egas.commandsEmergencyFuelCut to fuelSys.executesFuelCut {
                bind payload = egas.fuelCmdPort.fuelCmd;
                ref constraint <uc2_vc07>;
            }

            // [Sequence Flow]
            first msgInterventionReq_Stuck;

            // 공기량 제어가 불가능하므로, 연료 시스템으로만 차단 명령을 즉시 하달
            then msgFuelCutCmd;

            // [Verification Criterias]
            constraint <uc2_vc06> stuckResponseTime :> validationCriteria {
                doc
                /* 고착 감지 후 연료 차단 명령까지 10ms 이내 보장 */
                msgFuelCutCmd.start - msgInterventionReq_Stuck.done <= 10.0 [milli * s]
            }

            constraint <uc2_vc07> stuckFuelCutFailSafe :> validationCriteria {
                doc
                /* 페일세이프: 엔진 보호를 위한 연료 전면 차단 */
                msgFuelCutCmd.payload.injectionPulseWidth <= 0.0 [s]
            }

            constraint postcondition {
                doc
                /* 액추에이터 고착 대응 종료 후 연료 분사가 완전히 차단되어 토크가 안전 수준으로 저감된 상태여야 함 */
                msgFuelCutCmd.payload.injectionPulseWidth <= 0.0 [s]
            }
        }

        // ==========================================
        // 상호작용 및 시나리오 흐름(Flow) 정의
        // ==========================================

        // 1. 주 성공 시나리오(Normal)의 입출력 상호작용
        flow from addEcu to UC2_Normal_Emergency_Torque_Reduction;
        flow from UC2_Normal_Emergency_Torque_Reduction to engine;
        flow from UC2_Normal_Emergency_Torque_Reduction to fuelSys;

        // 2. 대안 시나리오(Alternative)의 입출력 상호작용
        flow from addEcu to UC2_Alternative_Communication_Delay_Compensation;
        flow from UC2_Alternative_Communication_Delay_Compensation to engine;
        flow from UC2_Alternative_Communication_Delay_Compensation to fuelSys;

        // 3. 예외 시나리오(Exception)의 입출력 상호작용 (스로틀 제어를 포기하므로 engine으로의 flow 생략)
        flow from addEcu to UC2_Exception_Actuator_Stuck_Fuel_Cut;
        flow from UC2_Exception_Actuator_Stuck_Fuel_Cut to fuelSys;
    }

    use case def <UC3> 'Hardware Fail-safe' {
        doc
        /* 이중화 센서 데이터를 실시간으로 모니터링하여 하드웨어 결함(온도 초과 등) 발생 시 즉각적인 페일세이프를 수행하고, 
         * 정상 작동 중에는 하드웨어 마찰 및 지연을 보정하여 실화를 방지하는 유스케이스
         */

        subject egas : EgasSystem;

        // 입력 액터
        actor sensors : AddSensorComponents;

        // 출력 및 피드백 액터
        actor engine : Engine;
        actor fuelSys : FuelInjectionIgnitionSystem;

        satisfy requirement 'SHR-03'; // reqFailSafeSensorMonitoring
        satisfy requirement 'SHR-02-2'; // reqMisfirePrevention

        // ==========================================
        // 액션(시나리오) 정의 - 각 액션이 하나의 완전한 시나리오를 대변함
        // ==========================================

        action UC3_Normal_Hardware_Delay_Compensation {
            doc
            /* [주 성공 시나리오] 
             * 양산 부품의 물리적 한계(마찰, 응답 지연 등)를 시스템이 실시간으로 자체 보정하여, 
             * 제어 명령 산출 후 액추에이터 지연을 10ms 이내로 통제하고 엔진 실화를 0Hz로 유지하는 전체 보정 제어 흐름
             */

            constraint precondition {
                doc
                /* 엔진 냉각수 온도가 120도 이하로, 시스템이 정상 구동 가능한 물리적 안전 한계 내에 있는 상태여야 함 */
                egas.sensorPort.sensorData.coolantTemp <= 120.0 ['°']
            }

            // [Messages]
            message msgNormalData of EngineSensorData
                from sensors.sendsNormalSensorData to egas.compensatesDelayAndFriction {
                bind payload = egas.sensorPort.sensorData;
            }

            message msgCompensatedAirCmd of IntakeAirFlow
                from egas.compensatesDelayAndFriction to engine.receivesCompensatedAir {
                bind payload = egas.airIntakePort.airFlow;
            }

            message msgCompensatedFuelCmd of FuelIgnitionCommand
                from egas.compensatesDelayAndFriction to fuelSys.executesCompensatedFuel {
                bind payload = egas.fuelCmdPort.fuelCmd;
                ref constraint <uc3_vc1>;
            }

            message msgCombustionFeedback of CombustionSystemStatus
                from fuelSys.sendsCombustionFeedback to egas.compensatesDelayAndFriction {
                bind payload = egas.combustionStatusPort.combustionStatus;
                ref constraint <uc3_vc2>;
            }

            // [Sequence Flow]
            first msgNormalData;

            // 정상 센서 데이터 수신 후 보정된 공기/연료 제어 명령을 병렬 하달
            fork commandOut;
                then msgCompensatedAirCmd;
                then msgCompensatedFuelCmd;
                    first msgCompensatedAirCmd then j;
                    first msgCompensatedFuelCmd then j;

            // 명령 수행 후 실화율 등 연소 상태 피드백 수신
            join j;
            then msgCombustionFeedback;

            // [Verification Criterias]
            constraint <uc3_vc1> actuatorLatencyConstraint :> validationCriteria {
                doc
                /* [VC#15] 응답 시간(EGAS 소유): 정상 센서 데이터 수신 후 보정된 연료/점화 명령 출력 시작까지 10ms 이내 */
                msgCompensatedFuelCmd.start - msgNormalData.done <= 10.0 [milli * s]
            }

            constraint <uc3_vc2> misfirePreventionConstraint :> validationCriteria {
                doc
                /* [VC#16] 실화 방지: 
                 * 피드백되는 연소 상태에서 엔진 실화(Misfire Rate)가 완벽히 억제(0Hz)되어야 함
                 */
                @PlantObservation;
                msgCombustionFeedback.payload.misfireRate <= 0.0 [Hz]
            }

            constraint postcondition {
                doc
                /* 보정 제어 종료 후 액추에이터 응답 지연이 10ms 이내이고 엔진 실화가 발생하지 않은 상태로 유지되어야 함 */
                msgCompensatedFuelCmd.start - msgNormalData.done <= 10.0 [milli * s] and
                msgCombustionFeedback.payload.misfireRate <= 0.0 [Hz]
            }
        }

        action UC3_Exception_Critical_Sensor_Fault_FailSafe {
            doc
            /* [예외 시나리오] 
             * 엔진 냉각수 온도가 120도를 초과하는 등 이중화 센서에 치명적 결함값이 입력될 경우, 
             * 급발진 및 엔진 파손을 막기 위해 즉각 공기 유입을 차단하고 연료 제어를 전면 중단하는 전체 페일세이프 흐름
             */
            constraint precondition {
                doc
                /* 엔진 냉각수 온도가 120도를 초과하는 등, 치명적인 시스템 파손이나 제어 불능이 우려되는 하드웨어 결함이 감지된 상태여야 함 */
                egas.sensorPort.sensorData.coolantTemp > 120.0 ['°']
            }

            // [Messages]
            message msgFaultyData of EngineSensorData
                from sensors.sendsFaultyData to egas.triggersEmergencyFailSafe {
                bind payload = egas.sensorPort.sensorData;
            }

            message msgAirCutCmd of IntakeAirFlow
                from egas.triggersEmergencyFailSafe to engine.receivesAirCut {
                bind payload = egas.airIntakePort.airFlow;
                ref constraint <uc3_vc3>;
            }

            message msgFuelCutCmd of FuelIgnitionCommand
                from egas.triggersEmergencyFailSafe to fuelSys.executesFuelCut {
                bind payload = egas.fuelCmdPort.fuelCmd;
                ref constraint <uc3_vc4>;
            }

            // [Sequence Flow]
            first msgFaultyData;

            // 결함 센서 데이터 수신 즉시 스로틀 공기량 차단과 연료 분사 차단 명령을 병렬 하달
            fork commandOut;
                then msgAirCutCmd;
                then msgFuelCutCmd;

            // [Verification Criterias]

            doc
            /* Invalidate current torque and enforce cutoff upon brake intervention */

            dependency from self to UC1::UC1_Exception_Brake_Override_Intervention;
            in baseTorque : TorqueInterventionRequest;
            in brake : BrakePedalPosition;
            out finalTorque : TorqueInterventionRequest;
            out isEmergencyCut : Boolean;

            constraint brakeOverridePriority {
                brake.pressure > 0.0 [Pa] implies isEmergencyCut == true
            } // Brake override priority constraint (SHR-01, VC#05)
        }

        action def ActuateEngineAndFuel {
            doc
            /* Air and fuel control commands based on final torque */

            dependency from self to UC1::UC1_Normal_Manual_Acceleration;
            dependency from self to UC1::UC1_Alternative_Cruise_Control_Driving;
            dependency from self to UC1::UC1_Exception_Brake_Override_Intervention;

            in finalTorque : TorqueInterventionRequest;
            in isEmergencyCut : Boolean;
            out airCmd : IntakeAirFlow;
            out fuelCmd : FuelIgnitionCommand;

            constraint emergencyCutoff {

            } // Failsafe cutoff constraint (VC#06, VC#07)
        }

        action def MonitorDynamicsAndFeedback {
            doc
            /* Monitor combustion status and vehicle dynamics feedback */

            dependency from self to UC1::UC1_Normal_Manual_Acceleration;
            dependency from self to UC1::UC1_Alternative_Cruise_Control_Driving;

            in combustionFeedback : CombustionSystemStatus;
            in dynamicsFeedback : VehicleDynamicsFeedback;

            // Precision and ride comfort constraints (SHR-02-1, VC#02-04)
            constraint steadyStatePrecision {
                @PlantObservation;
                combustionFeedback.steadyStateError <= 2.0 and
                combustionFeedback.torqueOvershoot <= 5.0
            }
            constraint rideComfort {
                @PlantObservation;
                dynamicsFeedback.longitudinalJerk <= 10.0
            }
        }

        action def ManageDrivingTorqueArbitration {
            doc
            /* UC1 integrated operational flow */

            // 외부 컨텍스트 모델과 매핑되는 I/O 파라미터
            in accelIn : AccelerationIntent;
            in ccCmdIn : CruiseControlCommand;
            in brakeIn : BrakePedalPosition;
            in combustionStatusIn : CombustionSystemStatus;
            in dynamicsStatusIn : VehicleDynamicsFeedback;

            out airFlowOut : IntakeAirFlow;
            out fuelCmdOut : FuelIgnitionCommand;

            action arbitrate : ArbitrateTorqueDemand;
            action overrideCheck : OverrideBrakeIntervention;
            action actuate : ActuateEngineAndFuel;
            action monitor : MonitorDynamicsAndFeedback;

            flow ccCmdIn to arbitrate.ccCmd;

            flow brakeIn to overrideCheck.brake;

            flow overrideCheck.isEmergencyCut to actuate.isEmergencyCut;

            flow actuate.airCmd to airFlowOut;
            flow actuate.fuelCmd to fuelCmdOut;

            flow combustionStatusIn to monitor.combustionFeedback;
            flow dynamicsStatusIn to monitor.dynamicsFeedback;

            constraint responseTime {

                (airFlowOut.responseTime <= 50.0 [milli * s] and
                    fuelCmdOut.responseTime <= 50.0 [milli * s])
            }
        }

    package EgasOperationalRequirements_UC2 {
        @MetaInformation {
            version = "v0.1";
        }

        // ==========================================
        // 1. 개별 운용 기능(Action) 정의
        // ==========================================
        action def ExecuteParallelTorqueReduction {
            doc
            /* [UC2_Normal] 15ms 이내 공기량 및 연료량 병렬 축소 */
            in torqueReq : TorqueInterventionRequest;
            out airCmd : IntakeAirFlow;
            out fuelCmd : FuelIgnitionCommand;

            // SHR-04-1, SHR-04-2 및 VC#01, VC#02, VC#03 제약 할당
            constraint rapidReduction {
                (torqueReq.requestedTorqueDelta < 0.0 [N * m] and
                    torqueReq.latency <= 15.0 [milli * s]) implies
                (airCmd.responseTime <= 15.0 [milli * s] and
                    fuelCmd.responseTime <= 15.0 [milli * s] and
                    airCmd.massFlow < airCmd.previousMassFlow and
                    fuelCmd.injectionPulseWidth < fuelCmd.previousPulseWidth)
            }
        }

        action def CompensateCommunicationDelay {
            doc
            /* [UC2_Alternative] 통신 지연(15ms 초과) 시 직전 제어값 유지 */
            in torqueReq : TorqueInterventionRequest;
            out airCmd : IntakeAirFlow;
            out fuelCmd : FuelIgnitionCommand;

            // VC#04, VC#05 제약 할당
            constraint maintainPreviousState {
                airCmd.massFlow == airCmd.previousMassFlow and
                fuelCmd.injectionPulseWidth == fuelCmd.previousPulseWidth
            }
        }

        action def MitigateActuatorStuck {
            doc
            /* [UC2_Exception] 스로틀 고착 감지 시 연료 전면 차단 (10ms 이내) */
            in airCmdFeedback : IntakeAirFlow;
            out fuelCmd : FuelIgnitionCommand;

            // VC#06, VC#07 제약 할당
            constraint failsafeFuelCut {
                airCmdFeedback.massFlow >= airCmdFeedback.previousMassFlow implies
                (fuelCmd.responseTime <= 10.0 [milli * s] and
                    fuelCmd.injectionPulseWidth <= 0.0 [s])
            }
        }

        // ==========================================
        // 2. 최상위 운용 시나리오 (Control & Object Flow)
        // ==========================================

        action def ManageEmergencyIntervention {
            doc
            /* UC2 전체의 판단 로직 및 데이터 흐름을 오케스트레이션 */

            in torqueReqIn : TorqueInterventionRequest;
            out airFlowOut : IntakeAirFlow;
            out fuelCmdOut : FuelIgnitionCommand;

            // 시나리오 분기 (Control Flow)
            if torqueReqIn.latency <= 15.0 [milli * s] {
                action normalReduction : ExecuteParallelTorqueReduction;
                flow torqueReqIn to normalReduction.torqueReq;

                // 하드웨어 고착 감지 판단을 위한 예외 처리 파이프라인
                action stuckMonitor : MitigateActuatorStuck;
                flow normalReduction.airCmd to stuckMonitor.airCmdFeedback;

                flow normalReduction.airCmd to airFlowOut;
                flow stuckMonitor.fuelCmd to fuelCmdOut;
            } else {
                action delayCompensator : CompensateCommunicationDelay;
                flow torqueReqIn to delayCompensator.torqueReq;

                flow delayCompensator.airCmd to airFlowOut;
                flow delayCompensator.fuelCmd to fuelCmdOut;
            }
        }
    } // package EgasOperationalRequirements_UC2

            doc
            /* [UC3_Normal] 정상 상태에서 하드웨어 마찰 및 지연을 보정하여 제어 명령 산출 */
            in sensorData : EngineSensorData;

            doc

            in faultySensorData : EngineSensorData;

            }

                airCutCmd.massFlow <= 0.0 [kg / s]
            }

            constraint sensorFaultFuelCutConstraint {
                fuelCutCmd.injectionPulseWidth <= 0.0 [s]
            }
        }

        // ==========================================
        // 2. 최상위 운용 시나리오 (Control & Object Flow)
        // ==========================================

            doc
            /* UC3 전체의 하드웨어 모니터링 판단 로직 및 데이터 흐름 오케스트레이션 */

            in sensorIn : EngineSensorData;
            in combustionStatusIn : CombustionSystemStatus;
            out airFlowOut : IntakeAirFlow;
            out fuelCmdOut : FuelIgnitionCommand;

            // 센서 데이터(냉각수 온도 등)를 기반으로 한 시나리오 분기 (Control Flow)
            if sensorIn.coolantTemp <= 120.0 ['°'] {

                action compensate : ExecuteDelayCompensation;

                action failSafe : ExecuteEmergencyFailSafe;

                flow sensorIn to failSafe.faultySensorData;

                flow failSafe.airCutCmd to airFlowOut;
                flow failSafe.fuelCutCmd to fuelCmdOut;
            }
        }

            in calReq : CalibrationRequest;

            in invalidReq : CalibrationRequest;

            }

            }
        }

        // ==========================================
        // 2. 최상위 운용 시나리오 (Control & Object Flow)
        // ==========================================

        action def ManagePowertrainCalibration {
            doc
            /* UC4 전체의 메모리 보안 검증 및 튜닝 데이터 업데이트 흐름 오케스트레이션 */

            in calReqIn : CalibrationRequest;
            out calStatusOut : CalibrationStatus;

            // 요청된 메모리 주소의 유효성에 따른 시나리오 분기 (Control Flow)
            if calReqIn.targetMemoryAddress > 0 {
                // [정상 흐름] 접근 인가 후 무결성 검증 수행
                action authorize : AuthorizeMemoryAccess;
                action verifyIntegrity : VerifyDataIntegrity;

                // 1) 접근 요청 처리 및 인가 응답
                flow calReqIn to authorize.calReq;
                flow authorize.calStatus to calStatusOut;

                // 2) 인가 완료 후 데이터 기록 및 무결성 검증으로 시퀀스 전이
                first authorize;
                then verifyIntegrity;

                // (실제 데이터 기록 요청이 들어왔다고 가정하고 검증 로직으로 데이터 흐름 연결)
                flow calReqIn to verifyIntegrity.writeReq;
                flow verifyIntegrity.calStatus to calStatusOut;
            } else {
                // [예외 흐름] 비유효 접근 즉각 거절 및 에러 반환
                action reject : RejectInvalidAccess;

                flow calReqIn to reject.invalidReq;
                flow reject.errorStatus to calStatusOut;
            }
        }
    } // package EgasOperationalRequirements_UC4

            // 요청된 진단 서비스 ID(serviceId)에 따른 제어 흐름(Control Flow) 다중 분기
            if diagReqIn.serviceId == 1 {
                // [정상 흐름 1] 고장 코드(DTC) 조회 처리
                action readDtc : ReadDiagnosticTroubleCode;

                flow sensorIn to readDtc.sensorData;
                flow diagReqIn to readDtc.diagReq;
                flow readDtc.diagStatus to diagStatusOut;
            } else if diagReqIn.serviceId == 2 {
                // [정상 흐름 2] 영점 조정 루틴 수행
                action zeroPointCal : ExecuteZeroPointCalibration;

                flow diagReqIn to zeroPointCal.diagReq;
                flow zeroPointCal.diagStatus to diagStatusOut;
            } else {
                // [예외 흐름] 비유효 서비스 ID 거절
                action rejectReq : RejectInvalidDiagnosticRequest;

                flow diagReqIn to rejectReq.diagReq;
                flow rejectReq.diagStatus to diagStatusOut;
            }
        }
    } // package EgasOperationalRequirements_UC5
} // package RV_OperationalModel

    state def DrivingControlRegion {
        first start then powerOff;

        state powerOff;
        state standby;

        state active {
            first start then manualDriving;
            state manualDriving {
                do action manageArbitration
                    : EgasOperationalRequirements_UC1::ManageDrivingTorqueArbitration;
            }
            state cruiseDriving {
                do action manageArbitration
                    : EgasOperationalRequirements_UC1::ManageDrivingTorqueArbitration;
            }

            transition t_manualToCruise
                first manualDriving accept CruiseControlCommand then cruiseDriving;
            transition t_cruiseToManual
                first cruiseDriving accept AccelerationIntent then manualDriving;
        }

        state brakeOverridden {
            do action overrideBrake : EgasOperationalRequirements_UC1::OverrideBrakeIntervention;
        }
        state waitForPedalRelease;

        transition t_standbyToActive first standby accept AccelerationIntent then active;
        transition t_activeToStandby first active then standby;
        transition t_activeToBrake first active accept BrakePedalPosition then brakeOverridden;
        transition t_brakeToWait first brakeOverridden then waitForPedalRelease;
        transition t_brakeToStandby first brakeOverridden then standby;
        transition t_waitToStandby first waitForPedalRelease then standby;
        transition t_powerOn first powerOff accept IgnitionState then standby;
        transition t_powerOffStandby first standby accept IgnitionState then powerOff;
        transition t_powerOffActive first active accept IgnitionState then powerOff;
        transition t_powerOffOverride first brakeOverridden accept IgnitionState then powerOff;
    }

    state def EmergencyInterventionRegion {
        first start then emergencyStandby;
        state emergencyStandby;

        state InterventionActive {
            first start then initialCheck;
            state initialCheck;
            state parallelReduction {
                do action manageEmergency
                    : EgasOperationalRequirements_UC2::ManageEmergencyIntervention;
            }
            state delayCompensating {
                do action manageEmergency
                    : EgasOperationalRequirements_UC2::ManageEmergencyIntervention;
            }

            // [래치] 해제 요청으로 나가지 않는다. 점화 OFF(t_ignitionOffReset)로만 해제한다.
            state actuatorStuckFallback {
                do action manageEmergency
                    : EgasOperationalRequirements_UC2::ManageEmergencyIntervention;
            }

            transition t_checkToNormal first initialCheck then parallelReduction;
            transition t_checkToDelay first initialCheck then delayCompensating;

            // [변경] 개입 중(delta < 0)일 때만 정상/지연 전환 → 해제 요청과 겹치지 않음
            transition t_normalToDelay
                first parallelReduction
                accept req : TorqueInterventionRequest
                if req.requestedTorqueDelta < 0.0 [N * m] and req.latency > 15.0 [milli * s]
                    then delayCompensating;
            transition t_delayToNormal
                first delayCompensating
                accept req : TorqueInterventionRequest
                if req.requestedTorqueDelta < 0.0 [N * m] and req.latency <= 15.0 [milli * s]
                    then parallelReduction;

            transition t_stuckDetect first parallelReduction then actuatorStuckFallback;

            // [변경] accept/가드 없는 완료 전이 → 해제 요청(delta >= 0)에 의한 전이
            transition t_finishNormal
                first parallelReduction
                accept req : TorqueInterventionRequest
                if req.requestedTorqueDelta >= 0.0 [N * m] then emergencyStandby;
            transition t_finishDelay
                first delayCompensating
                accept req : TorqueInterventionRequest
                if req.requestedTorqueDelta >= 0.0 [N * m] then emergencyStandby;
        }
        // 저감 요청(delta < 0)일 때만 개입 진입
        transition t_triggerIntervention
            first emergencyStandby
            accept req : TorqueInterventionRequest
            if req.requestedTorqueDelta < 0.0 [N * m] then InterventionActive;

        //  점화 OFF → 고착 래치를 포함한 모든 하위 상태 초기화
        transition t_ignitionOffReset
            first InterventionActive
            accept ign : IgnitionState
            if ign.status == IgnitionStatusEnum::Off then emergencyStandby;
    }

    state def SafetyMonitoringRegion {
        first start then normalMonitoring;
        state normalMonitoring {
            do action executeDelayComp : EgasOperationalRequirements_UC3::ExecuteDelayCompensation;
        }
        state criticalFault {
            do action executeFailSafe : EgasOperationalRequirements_UC3::ExecuteEmergencyFailSafe;
        }

        transition t_faultDetection
            first normalMonitoring accept EngineSensorData then criticalFault;
    }

    state def CalibrationHandling {
        first start then cal_Idle;
        state cal_Idle;
        state authorizing {
            do action authorize : EgasOperationalRequirements_UC4::AuthorizeMemoryAccess;
        }
        state verifying {
            do action verifyIntegrity : EgasOperationalRequirements_UC4::VerifyDataIntegrity;
        }
        state rejectingAndRollback {
            do action rejectInvalidAccess : EgasOperationalRequirements_UC4::RejectInvalidAccess;
        }

        transition t_reqAuth first cal_Idle then authorizing;
        transition t_invalidReq first cal_Idle then rejectingAndRollback;
        transition t_authToVerify first authorizing then verifying;
        transition t_authFail first authorizing then rejectingAndRollback;
        transition t_verifySuccess first verifying then cal_Idle;
        transition t_verifyFail first verifying then rejectingAndRollback;
        transition t_rejectToIdle first rejectingAndRollback then cal_Idle;
    }

    state def DiagnosticsHandling {
        first start then Diag_Idle;
        state Diag_Idle;
        state ReadingDTC {
            do action readDtc : EgasOperationalRequirements_UC5::ReadDiagnosticTroubleCode;
        }
        state CalibratingZeroPoint {
            do action zeroPointCal : EgasOperationalRequirements_UC5::ExecuteZeroPointCalibration;
        }
        state RejectingRequest {
            do action rejectReq : EgasOperationalRequirements_UC5::RejectInvalidDiagnosticRequest;
        }

        transition t_reqDTC first Diag_Idle then ReadingDTC;
        transition t_reqZero first Diag_Idle then CalibratingZeroPoint;
        transition t_reqInvalid first Diag_Idle then RejectingRequest;
        transition t_readToIdle first ReadingDTC then Diag_Idle;
        transition t_calToIdle first CalibratingZeroPoint then Diag_Idle;
        transition t_rejectToIdle first RejectingRequest then Diag_Idle;
    }

    state def EgasBehavior parallel {
        dependency
            self
            to
                EgasOperationalRequirements_UC1,
                EgasOperationalRequirements_UC2,
                EgasOperationalRequirements_UC3;

        // ==========================================
        // 1. 주행 토크 제어 영역 (UC1)

        // ==========================================

        // ==========================================

        state monitoringAndServiceRegion {

            state monitoringActive {

                state diagnosticsHandling : DiagnosticsHandling;
            }
        }
    }

package SensmetryRequirementTemplates {
    private import ScalarValues::*;
    doc
    /* Sensmetry Requirement Templates (공식 제약 조건 템플릿) */

    requirement def OverrideRequirement {
        doc
        /* When both {primaryInputActive} and {overrideInputActive} are true, the {outputValue} must be limited to {safeLimit}. */
        attribute primaryInputActive : Boolean;
        attribute overrideInputActive : Boolean;
        attribute outputValue : Real;
        attribute safeLimit : Real;

        assume constraint {
            safeLimit >= 0.0
        }
        require constraint {
            (primaryInputActive and overrideInputActive) implies outputValue <= safeLimit
        }
    }

    requirement def RedundancyDeviationRequirement {
        doc
        /* If the absolute difference between {channel1} and {channel2} exceeds {maxDeviation}, {faultFlag} must be asserted true. */
        attribute channel1 : Real;
        attribute channel2 : Real;
        attribute maxDeviation : Real;
        attribute faultFlag : Boolean;

        assume constraint {
            maxDeviation > 0.0
        }
        require constraint {
            (channel1 - channel2 > maxDeviation or channel2 - channel1 > maxDeviation) implies
            faultFlag == true
        }
    }

    requirement def PerformanceTimingRequirement {
        doc
        /* The {process} shall complete within {maxResponseTime}. */
        attribute executionTime : Real;
        attribute maxResponseTime : Real;

        assume constraint {
            maxResponseTime > 0.0
        }
        require constraint {
            executionTime <= maxResponseTime
        }
    }
} // package SensmetryRequirementTemplates
